/* v0.23: no LineClear executor / hole propagation.
   A boundary-closure wave is only a relation token.  Each occupied voxel sees

   Multiple closure waves superpose as repeated copies of the already learned
   +1 spatial effect; no row is selected or ordered. */
static float sf_cresp(const SharedField*f,int rel,int out){return sf_amp(sfget(f,kcresp(rel,out),.5));}
static void boundary_reach_one_side(const uint8_t*b,int dx,int dy,uint8_t wave[N]){
  memset(wave,0,N);
  for(int y=0;y<H;y++)for(int x=0;x<W;x++)if(b[y*W+x]){
    int px=x-dx,py=y-dy;
    if(px<0||px>=W||py<0||py>=H)wave[y*W+x]=1;
  }
  for(int step=0;step<N;step++){
    uint8_t next[N];memcpy(next,wave,N);int changed=0;
    for(int y=0;y<H;y++)for(int x=0;x<W;x++)if(wave[y*W+x]){
      int nx=x+dx,ny=y+dy;
      if(nx>=0&&nx<W&&ny>=0&&ny<H&&b[ny*W+nx]&&!next[ny*W+nx]){next[ny*W+nx]=1;changed=1;}
    }
    memcpy(wave,next,N);if(!changed)break;
  }
}

static int sf_clear_relation(SharedField*f,uint8_t*b){
  int dx[16],dy[16],nd=state_unit_dirs(b,dx,dy,16);uint8_t seed[N]={0};int any=0;
  for(int d=0;d<nd;d++){
    if(sf_amp(sfget(f,kclosure(dx[d],dy[d]),.5))<=0.f)continue;
    uint8_t q[N];if(!generic_boundary_wave(b,dx[d],dy[d],q))continue;
    for(int i=0;i<N;i++)seed[i]|=q[i];
    any=1;
  }
  if(!any)return 0;
  uint8_t before[N],out[N]={0};memcpy(before,b,N);
  for(int sy=0;sy<H;sy++)for(int x=0;x<W;x++)if(before[sy*W+x]){
    double survive=1.0,shift=0.0;
    for(int cy=0;cy<H;cy++)if(seed[cy*W+x]){
      int rel=addr_vertical_relation(sy,cy);
      survive*=1.0-(double)sf_cresp(f,rel,2);
      shift+=(double)sf_cresp(f,rel,1);
    }
    if(survive<=.5)continue;

    if(oy>=0&&oy<H)out[oy*W+x]=1;
  }
  memcpy(b,out,N);return !eq(before,b,N);
}

/* v0.22: geometry only proposes destinations.  Compatibility is not hard-coded.
   A proposal wave collides with a destination-state token (empty/occupied/boundary),
   and the learned compatibility amplitude from the same shared field determines
   whether the multi-voxel carrier survives. */

  memset(out,0,N);int tx[64],ty[64],n=sf_geom_targets(f,act,a,tx,ty,64);if(n<0)return 0.f;
  float C=1.f;
  for(int i=0;i<n;i++){

    else{state=w[ty[i]*W+tx[i]]?1:0;out[ty[i]*W+tx[i]]=1;}

  }

/* v0.23: no LineClear executor / hole propagation.
   A boundary-closure wave is only a relation token.  Each occupied voxel sees
   the relative vertical address to every closure token in its own column.  The
   shared field says whether that relation means persist / +1-down / disappear.
   Multiple closure waves superpose as repeated copies of the already learned
   +1 spatial effect; no row is selected or ordered. */
static float sf_cresp(const SharedField*f,int rel,int out){return sf_amp(sfget(f,kcresp(rel,out),.5));}
static void boundary_reach_one_side(const uint8_t*b,int dx,int dy,uint8_t wave[N]){
  memset(wave,0,N);
  for(int y=0;y<H;y++)for(int x=0;x<W;x++)if(b[y*W+x]){
    int px=x-dx,py=y-dy;
    if(px<0||px>=W||py<0||py>=H)wave[y*W+x]=1;
  }
  for(int step=0;step<N;step++){
    uint8_t next[N];memcpy(next,wave,N);int changed=0;
    for(int y=0;y<H;y++)for(int x=0;x<W;x++)if(wave[y*W+x]){
      int nx=x+dx,ny=y+dy;
      if(nx>=0&&nx<W&&ny>=0&&ny<H&&b[ny*W+nx]&&!next[ny*W+nx]){next[ny*W+nx]=1;changed=1;}
    }
    memcpy(wave,next,N);if(!changed)break;
  }
}
static int generic_boundary_wave(const uint8_t*b,int dx,int dy,uint8_t reached[N]){
  uint8_t a[N],z[N];boundary_reach_one_side(b,dx,dy,a);boundary_reach_one_side(b,-dx,-dy,z);
  int any=0;for(int i=0;i<N;i++){reached[i]=(uint8_t)(a[i]&&z[i]);any|=reached[i];}
  return any;
}
static int sf_clear_relation(SharedField*f,uint8_t*b){
  int dx[16],dy[16],nd=state_unit_dirs(b,dx,dy,16);uint8_t seed[N]={0};int any=0;
  for(int d=0;d<nd;d++){
    if(sf_amp(sfget(f,kclosure(dx[d],dy[d]),.5))<=0.f)continue;
    uint8_t q[N];if(!generic_boundary_wave(b,dx[d],dy[d],q))continue;
    for(int i=0;i<N;i++)seed[i]|=q[i];
    any=1;
  }
  if(!any)return 0;
  uint8_t before[N],out[N]={0};memcpy(before,b,N);
  for(int sy=0;sy<H;sy++)for(int x=0;x<W;x++)if(before[sy*W+x]){
    double survive=1.0,shift=0.0;
    for(int cy=0;cy<H;cy++)if(seed[cy*W+x]){
      int rel=addr_vertical_relation(sy,cy);
      survive*=1.0-(double)sf_cresp(f,rel,2);
      shift+=(double)sf_cresp(f,rel,1);
    }
    if(survive<=.5)continue;
    int oy=sy+(int)llround(shift);
    if(oy>=0&&oy<H)out[oy*W+x]=1;
  }
  memcpy(b,out,N);return !eq(before,b,N);
}

/* Apply one anonymous address-relation x effect hypothesis to the current mid-world.
   rel: 0=same address, 1=closure below source, 2=closure above source.
   effect: 1=reuse +1 spatial effect, 2=disappear.  Persistence is the default
   no-effect case and therefore does not need its own learned rule. */
static void sf_apply_closure_effect_hyp(const uint8_t before[N],const uint8_t seed[N],int rel,int effect,uint8_t pred[N]){
  memcpy(pred,before,N);
  uint8_t affected[N]={0};int shiftcount[N]={0};
  for(int sy=0;sy<H;sy++)for(int x=0;x<W;x++)if(before[sy*W+x]){
    int i=sy*W+x;
    for(int cy=0;cy<H;cy++)if(seed[cy*W+x]&&addr_vertical_relation(sy,cy)==rel){affected[i]=1;if(effect==1)shiftcount[i]++;}
  }
  if(effect==2){for(int i=0;i<N;i++)if(affected[i])pred[i]=0;return;}
  if(effect==1){

   Multiple closure waves superpose as repeated copies of the already learned
   +1 spatial effect; no row is selected or ordered. */
static float sf_cresp(const SharedField*f,int rel,int out){return sf_amp(sfget(f,kcresp(rel,out),.5));}
static void boundary_reach_one_side(const uint8_t*b,int dx,int dy,uint8_t wave[N]){
  memset(wave,0,N);
  for(int y=0;y<H;y++)for(int x=0;x<W;x++)if(b[y*W+x]){
    int px=x-dx,py=y-dy;
    if(px<0||px>=W||py<0||py>=H)wave[y*W+x]=1;
  }
  for(int step=0;step<N;step++){
    uint8_t next[N];memcpy(next,wave,N);int changed=0;
    for(int y=0;y<H;y++)for(int x=0;x<W;x++)if(wave[y*W+x]){
      int nx=x+dx,ny=y+dy;
      if(nx>=0&&nx<W&&ny>=0&&ny<H&&b[ny*W+nx]&&!next[ny*W+nx]){next[ny*W+nx]=1;changed=1;}
    }
    memcpy(wave,next,N);if(!changed)break;
  }
}
static int generic_boundary_wave(const uint8_t*b,int dx,int dy,uint8_t reached[N]){
  uint8_t a[N],z[N];boundary_reach_one_side(b,dx,dy,a);boundary_reach_one_side(b,-dx,-dy,z);

  return any;
}
static int sf_clear_relation(SharedField*f,uint8_t*b){
  int dx[16],dy[16],nd=state_unit_dirs(b,dx,dy,16);uint8_t seed[N]={0};int any=0;
  for(int d=0;d<nd;d++){
    if(sf_amp(sfget(f,kclosure(dx[d],dy[d]),.5))<=0.f)continue;
    uint8_t q[N];if(!generic_boundary_wave(b,dx[d],dy[d],q))continue;
    for(int i=0;i<N;i++)seed[i]|=q[i];
    any=1;
  }
  if(!any)return 0;
  uint8_t before[N],out[N]={0};memcpy(before,b,N);
  for(int sy=0;sy<H;sy++)for(int x=0;x<W;x++)if(before[sy*W+x]){
    double survive=1.0,shift=0.0;
    for(int cy=0;cy<H;cy++)if(seed[cy*W+x]){
      int rel=addr_vertical_relation(sy,cy);
      survive*=1.0-(double)sf_cresp(f,rel,2);
      shift+=(double)sf_cresp(f,rel,1);
    }
    if(survive<=.5)continue;
    int oy=sy+(int)llround(shift);
    if(oy>=0&&oy<H)out[oy*W+x]=1;
  }
  memcpy(b,out,N);return !eq(before,b,N);
}

   its own one-micro mid state.  Only the residual mid->final is allowed to give

/* Apply one anonymous address-relation x effect hypothesis to the current mid-world.
   rel: 0=same address, 1=closure below source, 2=closure above source.
   effect: 1=reuse +1 spatial effect, 2=disappear.  Persistence is the default
   no-effect case and therefore does not need its own learned rule. */
static void sf_apply_closure_effect_hyp(const uint8_t before[N],const uint8_t seed[N],int rel,int effect,uint8_t pred[N]){
  memcpy(pred,before,N);
  uint8_t affected[N]={0};int shiftcount[N]={0};
  for(int sy=0;sy<H;sy++)for(int x=0;x<W;x++)if(before[sy*W+x]){
    int i=sy*W+x;
    for(int cy=0;cy<H;cy++)if(seed[cy*W+x]&&addr_vertical_relation(sy,cy)==rel){affected[i]=1;if(effect==1)shiftcount[i]++;}
  }
  if(effect==2){for(int i=0;i<N;i++)if(affected[i])pred[i]=0;return;}
  if(effect==1){
    uint8_t out[N];memcpy(out,before,N);
    for(int i=0;i<N;i++)if(affected[i])out[i]=0;
    for(int sy=0;sy<H;sy++)for(int x=0;x<W;x++){int i=sy*W+x;if(!affected[i])continue;int oy=sy+shiftcount[i];if(oy>=0&&oy<H)out[oy*W+x]=1;}
    memcpy(pred,out,N);
  }
}

   visible mid->final residual.  No per-voxel identity label, no clear-row target,
   no persist/down/gone annotation is exposed. */
static void sf_train_closure_from_visible_transitions(SharedField*f,int events){
  RNG r={0xC10532ULL};
  for(int q=0;q<events;q++){
    int act=ri(&r,ACTS);State before,truth,mid;
    if(!place_piece(&r,&before,act,1)){q--;continue;}
    truth=before;(void)real_step(&truth,act);sf_predict_mid_no_downstream(f,&before,act,&mid);
    int e0=world_hdist(mid.world,truth.world);
    int dirs_x[16],dirs_y[16],nd=state_unit_dirs(mid.world,dirs_x,dirs_y,16);
    for(int d=0;d<nd;d++){
      uint8_t seed[N];if(!generic_boundary_wave(mid.world,dirs_x[d],dirs_y[d],seed))continue;
      int changed=e0>0;sfcredit(f,kclosure(dirs_x[d],dirs_y[d]),1.0,changed);
      if(!changed)continue;
      for(int rel=0;rel<3;rel++){
        double gain[3]={0,0,0};
        for(int effect=1;effect<=2;effect++){
          uint8_t pred[N];sf_apply_closure_effect_hyp(mid.world,seed,rel,effect,pred);
          int e1=world_hdist(pred,truth.world);
          gain[effect]=e0>e1?(double)(e0-e1)/(double)e0:0.0;
        }
        /* Continuous residual competition between alternative consequences.
           No top-1/winner selection: each effect receives its own explanatory
           mass as positive credit and the alternative's mass as negative credit. */
        if(gain[1]>0.0||gain[2]>0.0){
          sfcredit(f,kcresp(rel,1),gain[1],1);sfcredit(f,kcresp(rel,1),gain[2],0);
          sfcredit(f,kcresp(rel,2),gain[2],1);sfcredit(f,kcresp(rel,2),gain[1],0);
        }
      }
    }
  }
}
