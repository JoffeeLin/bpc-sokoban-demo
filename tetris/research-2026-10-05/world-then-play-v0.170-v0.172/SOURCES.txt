/* BPC World-Then-Play v0.170
   Stage A: freeze/re-audit v0.153 as the learned world dynamics.
   Stage B baseline: learn Screen -> operation only from full 10x20 visual
   demonstrations.  The policy learner has no reward/score/search/board
   heuristic; the external teacher is used only to generate observed actions.
*/

#define FW 17
#define FH 22
#define OP_HT_BITS 20
#define OP_HT_SIZE (1u<<OP_HT_BITS)
#define OP_HT_MASK (OP_HT_SIZE-1)
#define MAXVF 4096
#define TRAIN_DECISIONS 1500
#define TEST_DECISIONS 400
#define PLAY_TICKS 1500

typedef struct { uint8_t p[FH][FW]; } VFrame;
typedef struct { uint64_t key; float c[4]; float total; } OpEntry;
typedef struct { OpEntry *tab; int n; float base[4], total; } OpField;

static void vrender(const State*s,VFrame*f){
    memset(f,0,sizeof(*f));
    /* four visible reference corners; they are pixels, not hidden coordinates */
    f->p[0][0]=3; f->p[0][FW-1]=3; f->p[FH-1][0]=3; f->p[FH-1][FW-1]=3;
    for(int y=0;y<s->h && y<20;y++)for(int x=0;x<s->w && x<10;x++){
        if(s->world[y][x]) f->p[y+1][x+1]=1;      /* settled visual color */
        if(s->a[y][x])     f->p[y+1][x+1]=2;      /* moving visual color */
    }
    /* preview is rendered in a side box, using the SAME moving-block color. */
    for(int y=0;y<4;y++)for(int x=0;x<4;x++)if(s->preview[y][x])f->p[y+1][x+12]=2;
}

/* This decoder is used ONLY for the Stage-A information-closure audit.
   It is never used by the operation learner. */
static void vdecode_audit(const VFrame*f,State*s){
    zero(s,TEST_W,TEST_H);
    for(int y=0;y<20;y++)for(int x=0;x<10;x++){
        uint8_t q=f->p[y+1][x+1];
        if(q==1)s->world[y][x]=1;
        else if(q==2)s->a[y][x]=1;
    }
    for(int y=0;y<4;y++)for(int x=0;x<4;x++)if(f->p[y+1][x+12]==2)s->preview[y][x]=1;
}

static int visual_info_closure_audit(int nsteps){
    State s,n,d; reset_game(&s); int ok=0,total=0; const int amap[4]={0,1,2,3};
    for(int t=0;t<nsteps;t++){
        VFrame vf; vrender(&s,&vf); vdecode_audit(&vf,&d); ok+=same(&s,&d); total++;
        int a=ri(4); reality(&s,a,amap,&n);
        s=n;
        if(last_truth_spawn_attempt){ if(last_truth_spawn_success)refresh_preview(&s); else reset_game(&s); }
    }
    printf("STAGE_A visual_information_closure=%d/%d\n",ok,total);
    return ok==total;
}

static uint64_t op_mix(uint64_t x){return mix64(x)|1ULL;}
static uint64_t relkey(int c1,int c2,int dx,int dy){
    uint64_t z=0x6a09e667f3bcc909ULL;
    z^=mix64((uint64_t)(c1&7)+0x10ULL);
    z^=mix64((uint64_t)(c2&7)+0x30ULL);
    z^=mix64((uint64_t)(dx+32)+0x100ULL);
    z^=mix64((uint64_t)(dy+32)+0x300ULL);
    return op_mix(z);
}
static uint64_t pairkey(uint64_t a,uint64_t b){if(a>b){uint64_t t=a;a=b;b=t;}return op_mix(a^mix64(b+0x9e3779b97f4a7c15ULL));}
static int addf(uint64_t fs[],int*n,uint64_t k){if(!k||*n>=MAXVF)return 0;for(int i=0;i<*n;i++)if(fs[i]==k)return 1;fs[(*n)++]=k;return 1;}

static int vfeatures(const VFrame*f,uint64_t fs[]){
    int xs[256],ys[256],cs[256],nn=0,n=0;
    for(int y=0;y<FH;y++)for(int x=0;x<FW;x++)if(f->p[y][x] && nn<256){xs[nn]=x;ys[nn]=y;cs[nn]=f->p[y][x];nn++;}
    /* D1: generic visible color relations.  Local world/world relations plus
       all relations involving a moving-color or visible reference pixel. */
    uint64_t anchor_rel[256][24]; int ar_n[256]; memset(ar_n,0,sizeof(ar_n));
    for(int i=0;i<nn;i++){
        for(int j=0;j<nn;j++)if(i!=j){
            int dx=xs[j]-xs[i],dy=ys[j]-ys[i];
            int local=(abs(dx)<=5 && abs(dy)<=5);
            int informative=(cs[i]==2 || cs[j]==2 || cs[i]==3 || cs[j]==3);
            if(!local && !informative)continue;
            uint64_t r=relkey(cs[i],cs[j],dx,dy); addf(fs,&n,r);
            if(ar_n[i]<24)anchor_rel[i][ar_n[i]++]=r;
        }
    }
    /* D2: ordinary collision of two relations sharing the same visible pixel. */
    for(int i=0;i<nn;i++){
        int m=ar_n[i];
        for(int a=0;a<m;a++)for(int b=a+1;b<m;b++){
            if(n>=MAXVF) break;
            addf(fs,&n,pairkey(anchor_rel[i][a],anchor_rel[i][b]));
        }
        if(n>=MAXVF)break;
    }
    return n;
}

static void op_init(OpField*f){memset(f,0,sizeof(*f));f->tab=calloc(OP_HT_SIZE,sizeof(OpEntry));if(!f->tab){fprintf(stderr,"op alloc\n");exit(2);}}
static void op_free(OpField*f){free(f->tab);}
static OpEntry*op_find(OpField*f,uint64_t k,int create){uint32_t h=(uint32_t)mix64(k)&OP_HT_MASK;for(uint32_t i=0;i<OP_HT_SIZE;i++){OpEntry*e=&f->tab[(h+i)&OP_HT_MASK];if(e->key==k)return e;if(!e->key){if(!create)return NULL;e->key=k;f->n++;return e;}}return NULL;}
static void op_learn(OpField*f,const VFrame*vf,int action){uint64_t fs[MAXVF];int nf=vfeatures(vf,fs);f->base[action]+=1;f->total+=1;for(int i=0;i<nf;i++){OpEntry*e=op_find(f,fs[i],1);if(!e)continue;e->c[action]+=1;e->total+=1;}}
static int op_predict(OpField*f,const VFrame*vf,double score[4]){
    uint64_t fs[MAXVF];int nf=vfeatures(vf,fs);for(int a=0;a<4;a++)score[a]=(f->base[a]+1.0)/(f->total+4.0)-0.25;
    double z[4]={0,0,0,0};
    for(int i=0;i<nf;i++){
        OpEntry*e=op_find(f,fs[i],0);if(!e||e->total<2)continue;double conf=e->total/(e->total+4.0);
        for(int a=0;a<4;a++){double p=(e->c[a]+1.0)/(e->total+4.0);double d=(p-0.25)*conf;score[a]+=d*fabs(d);z[a]+=fabs(d);}
    }
    int best=0;double bs=-1e99;for(int a=0;a<4;a++){double s=z[a]>0?score[a]/z[a]:score[a];if(s>bs){bs=s;best=a;}}
    return best; /* external measurement boundary */
}

static int min_active_x(const State*s){int m=s->w;for(int y=0;y<s->h;y++)for(int x=0;x<s->w;x++)if(s->a[y][x]&&x<m)m=x;return m;}
static int max_active_x(const State*s){int m=-1;for(int y=0;y<s->h;y++)for(int x=0;x<s->w;x++)if(s->a[y][x]&&x>m)m=x;return m;}
static int board_holes(const State*s){int holes=0;for(int x=0;x<s->w;x++){int seen=0;for(int y=0;y<s->h;y++){if(s->world[y][x])seen=1;else if(seen)holes++;}}return holes;}
static int col_height(const State*s,int x){for(int y=0;y<s->h;y++)if(s->world[y][x])return s->h-y;return 0;}
static double board_score(const State*s,int clears){double agg=0,bump=0,maxh=0;for(int x=0;x<s->w;x++){int h=col_height(s,x);agg+=h;if(h>maxh)maxh=h;if(x)bump+=fabs((double)h-col_height(s,x-1));}int holes=board_holes(s);return 0.55*agg+1.8*holes+0.35*bump+0.8*maxh-8.0*clears;}

static int apply_phys_once(const State*s,int op,State*out){const int amap[4]={0,1,2,3};unsigned long long save=truth_clear_events;reality(s,op,amap,out);int changed=!same(s,out);truth_clear_events=save;return changed;}
static int drop_one_piece(const State*s,State*out,int*clears){const int amap[4]={0,1,2,3};State x=*s,y;unsigned long long c0=truth_clear_events;for(int k=0;k<64;k++){reality(&x,2,amap,&y);x=y;if(last_truth_spawn_attempt){*out=x;*clears=(int)(truth_clear_events-c0);return 1;}}*out=x;*clears=(int)(truth_clear_events-c0);return 0;}

/* External demonstrator only.  It may use search/heuristic; BPC never sees
   these quantities, only the resulting visible action. */
static int teacher_action(const State*s){
    double best=1e99;int best_r=0,best_shift=0;int curx=min_active_x(s);
    for(int r=0;r<4;r++){
        State sr=*s,tmp;int valid=1;
        for(int k=0;k<r;k++){apply_phys_once(&sr,3,&tmp);if(same(&sr,&tmp)&&k==0){/* symmetric piece is still valid */}sr=tmp;}
        int mn=min_active_x(&sr),mx=max_active_x(&sr);if(mn<0||mx<0)continue;
        for(int target=0;target<=sr.w-1-(mx-mn);target++){
            State sh=sr;int delta=target-mn;int op=delta<0?0:1;for(int k=0;k<abs(delta);k++){apply_phys_once(&sh,op,&tmp);if(same(&sh,&tmp)){valid=0;break;}sh=tmp;}if(!valid){valid=1;continue;}
            State end;int clears=0;if(!drop_one_piece(&sh,&end,&clears))continue;double sc=board_score(&end,clears);
            if(sc<best){best=sc;best_r=r;best_shift=target-curx;}
        }
    }
    if(best_r>0)return 3;
    if(best_shift<0)return 0;
    if(best_shift>0)return 1;
    return 2;
}

static void game_apply(State*s,int a,long long*locks,long long*clears,long long*gameovers){const int amap[4]={0,1,2,3};State n;unsigned long long cb=truth_clear_events;reality(s,a,amap,&n);int attempt=last_truth_spawn_attempt,success=last_truth_spawn_success;*clears+=(long long)(truth_clear_events-cb);if(attempt)(*locks)++;*s=n;if(attempt&&success)refresh_preview(s);else if(attempt&&!success){(*gameovers)++;reset_game(s);}}

static void train_operation(OpField*f,int decisions){State s;reset_game(&s);long long L=0,C=0,G=0;for(int i=0;i<decisions;i++){int a=teacher_action(&s);VFrame vf;vrender(&s,&vf);op_learn(f,&vf,a);game_apply(&s,a,&L,&C,&G);}printf("STAGE_B train decisions=%d entries=%d teacher_locks=%lld clears=%lld gameovers=%lld\n",decisions,f->n,L,C,G);}
static void eval_operation(OpField*f,int decisions){State s;reset_game(&s);long long ok=0;for(int i=0;i<decisions;i++){int ta=teacher_action(&s);VFrame vf;double sc[4];vrender(&s,&vf);int pa=op_predict(f,&vf,sc);ok+=pa==ta;long long z=0;game_apply(&s,ta,&z,&z,&z);}printf("STAGE_B heldout_action_exact=%lld/%d (%.2f%%)\n",ok,decisions,100.0*ok/decisions);}
static void eval_autoplay(OpField*f,int ticks,const char*tag){State s;reset_game(&s);long long L=0,C=0,G=0,match=0;for(int i=0;i<ticks;i++){int ta=teacher_action(&s);VFrame vf;double sc[4];vrender(&s,&vf);int a=op_predict(f,&vf,sc);match+=a==ta;game_apply(&s,a,&L,&C,&G);}printf("STAGE_B %s ticks=%d teacher_match=%lld/%d locks=%lld clears=%lld gameovers=%lld\n",tag,ticks,match,ticks,L,C,G);}

int main(void){
    puts("BPC World-Then-Play v0.170 diagnostic"); fflush(stdout);
    rs=0x123456789abcdefULL;
    visual_info_closure_audit(10000); fflush(stdout);

    /* Stage A: reproduce/freeze the complete world model before policy work. */
    {Field *wf=calloc(1,sizeof(Field));const int amap[4]={0,1,2,3};rs=0x9e3779b97f4a7c15ULL;train(wf,amap,0);long long s8=eval_step(wf,amap,0,8,NORMAL),p8=eval_prog(wf,amap,0,8,NORMAL),lc=eval_lineclear(wf,amap,0),sp=eval_spawn(wf,amap,0),go=eval_gameover(wf,amap,0);RolloutStat rr=eval_full_rollout(wf,amap,0,20000);printf("STAGE_A frozen_world step8=%lld/%d program8=%lld/%d lineclear=%lld/%d spawn=%lld/%d gameover=%lld/%d rollout=%lld/%lld K=%.2f\n",s8,STEP_TESTS,p8,PROG_TESTS,lc,PROG_TESTS,sp,PROG_TESTS,go,PROG_TESTS,rr.ok,rr.ticks,Keff(wf));free(wf); fflush(stdout);}

    /* Stage B baseline: the world is frozen; learn only Screen -> Action. */
    OpField op;op_init(&op);rs=0xd1b54a32d192ed03ULL;train_operation(&op,TRAIN_DECISIONS); fflush(stdout);rs=0x94d049bb133111ebULL;eval_operation(&op,TEST_DECISIONS); fflush(stdout);rs=0xbf58476d1ce4e5b9ULL;eval_autoplay(&op,PLAY_TICKS,"autoplay");op_free(&op);
    return 0;
}

            State sh=sr;int delta=target-mn;int op=delta<0?0:1;for(int k=0;k<abs(delta);k++){apply_phys_once(&sh,op,&tmp);if(same(&sh,&tmp)){valid=0;break;}sh=tmp;}if(!valid){valid=1;continue;}
            State end;int clears=0;if(!drop_one_piece(&sh,&end,&clears))continue;double sc=board_score(&end,clears);
            if(sc<best){best=sc;best_r=r;best_shift=target-curx;}
        }
    }
    if(best_r>0)return 3;
    if(best_shift<0)return 0;
    if(best_shift>0)return 1;
    return 2;
}

static void game_apply(State*s,int a,long long*locks,long long*clears,long long*gameovers){const int amap[4]={0,1,2,3};State n;unsigned long long cb=truth_clear_events;reality(s,a,amap,&n);int attempt=last_truth_spawn_attempt,success=last_truth_spawn_success;*clears+=(long long)(truth_clear_events-cb);if(attempt)(*locks)++;*s=n;if(attempt&&success)refresh_preview(s);else if(attempt&&!success){(*gameovers)++;reset_game(s);}}

static void train_operation(OpField*f,int decisions){State s;reset_game(&s);long long L=0,C=0,G=0;for(int i=0;i<decisions;i++){int a=teacher_action(&s);VFrame vf;vrender(&s,&vf);op_learn(f,&vf,a);game_apply(&s,a,&L,&C,&G);}printf("STAGE_B train decisions=%d entries=%d teacher_locks=%lld clears=%lld gameovers=%lld\n",decisions,f->n,L,C,G);}
static void eval_operation(OpField*f,int decisions){State s;reset_game(&s);long long ok=0;for(int i=0;i<decisions;i++){int ta=teacher_action(&s);VFrame vf;double sc[4];vrender(&s,&vf);int pa=op_predict(f,&vf,sc);ok+=pa==ta;long long z=0;game_apply(&s,ta,&z,&z,&z);}printf("STAGE_B heldout_action_exact=%lld/%d (%.2f%%)\n",ok,decisions,100.0*ok/decisions);}
static void eval_autoplay(OpField*f,int ticks,const char*tag){State s;reset_game(&s);long long L=0,C=0,G=0,match=0;for(int i=0;i<ticks;i++){int ta=teacher_action(&s);VFrame vf;double sc[4];vrender(&s,&vf);int a=op_predict(f,&vf,sc);match+=a==ta;game_apply(&s,a,&L,&C,&G);}printf("STAGE_B %s ticks=%d teacher_match=%lld/%d locks=%lld clears=%lld gameovers=%lld\n",tag,ticks,match,ticks,L,C,G);}


static void eval_teacher_play(int ticks,const char*tag){State s;reset_game(&s);long long L=0,C=0,G=0;for(int i=0;i<ticks;i++){int a=teacher_action(&s);game_apply(&s,a,&L,&C,&G);}printf("TEACHER %s ticks=%d locks=%lld clears=%lld gameovers=%lld\n",tag,ticks,L,C,G);}

static void onpolicy_action_residual(OpField*f,int ticks,int round){
    State s;reset_game(&s);long long L=0,C=0,G=0,match=0;
    for(int i=0;i<ticks;i++){
        VFrame vf;double sc[4];vrender(&s,&vf);int pa=op_predict(f,&vf,sc);int ta=teacher_action(&s);
        match+=pa==ta;
        /* teacher action is an observed operation consequence at the actually
           visited state; only the action residual is written back. */
        op_learn(f,&vf,ta);
        game_apply(&s,pa,&L,&C,&G);
    }
    printf("ACTION_RESIDUAL round=%d visited=%d preupdate_match=%lld/%d locks=%lld clears=%lld gameovers=%lld entries=%d\n",round,ticks,match,ticks,L,C,G,f->n);
}

#define main v153_embedded_main

/* BPC World-Then-Play v0.172
   Stage A: freeze/re-audit v0.153 as the learned world dynamics.
   Stage B baseline: learn Screen -> operation only from full 10x20 visual
   demonstrations.  The policy learner has no reward/score/search/board
   heuristic; the external teacher is used only to generate observed actions.
*/

#define FW 17
#define FH 22
#define OP_HT_BITS 20
#define OP_HT_SIZE (1u<<OP_HT_BITS)
#define OP_HT_MASK (OP_HT_SIZE-1)
#define MAXVF 4096
#define TRAIN_DECISIONS 1500
#define TEST_DECISIONS 400
#define PLAY_TICKS 1500

typedef struct { uint8_t p[FH][FW]; } VFrame;
typedef struct { uint64_t key; float c[4]; float total; } OpEntry;
typedef struct { OpEntry *tab; int n; float base[4], total; } OpField;

static void vrender(const State*s,VFrame*f){
    memset(f,0,sizeof(*f));
    /* four visible reference corners; they are pixels, not hidden coordinates */
    f->p[0][0]=3; f->p[0][FW-1]=3; f->p[FH-1][0]=3; f->p[FH-1][FW-1]=3;
    for(int y=0;y<s->h && y<20;y++)for(int x=0;x<s->w && x<10;x++){
        if(s->world[y][x]) f->p[y+1][x+1]=1;      /* settled visual color */
        if(s->a[y][x])     f->p[y+1][x+1]=2;      /* moving visual color */
    }
    /* preview is rendered in a side box, using the SAME moving-block color. */
    for(int y=0;y<4;y++)for(int x=0;x<4;x++)if(s->preview[y][x])f->p[y+1][x+12]=2;
}

/* This decoder is used ONLY for the Stage-A information-closure audit.
   It is never used by the operation learner. */
static void vdecode_audit(const VFrame*f,State*s){
    zero(s,TEST_W,TEST_H);
    for(int y=0;y<20;y++)for(int x=0;x<10;x++){
        uint8_t q=f->p[y+1][x+1];
        if(q==1)s->world[y][x]=1;
        else if(q==2)s->a[y][x]=1;
    }
    for(int y=0;y<4;y++)for(int x=0;x<4;x++)if(f->p[y+1][x+12]==2)s->preview[y][x]=1;
}

static int visual_info_closure_audit(int nsteps){
    State s,n,d; reset_game(&s); int ok=0,total=0; const int amap[4]={0,1,2,3};
    for(int t=0;t<nsteps;t++){
        VFrame vf; vrender(&s,&vf); vdecode_audit(&vf,&d); ok+=same(&s,&d); total++;
        int a=ri(4); reality(&s,a,amap,&n);
        s=n;
        if(last_truth_spawn_attempt){ if(last_truth_spawn_success)refresh_preview(&s); else reset_game(&s); }
    }
    printf("STAGE_A visual_information_closure=%d/%d\n",ok,total);
    return ok==total;
}

static uint64_t op_mix(uint64_t x){return mix64(x)|1ULL;}
static uint64_t relkey(int c1,int c2,int dx,int dy){
    uint64_t z=0x6a09e667f3bcc909ULL;
    z^=mix64((uint64_t)(c1&7)+0x10ULL);
    z^=mix64((uint64_t)(c2&7)+0x30ULL);
    z^=mix64((uint64_t)(dx+32)+0x100ULL);
    z^=mix64((uint64_t)(dy+32)+0x300ULL);
    return op_mix(z);
}
static uint64_t pairkey(uint64_t a,uint64_t b){if(a>b){uint64_t t=a;a=b;b=t;}return op_mix(a^mix64(b+0x9e3779b97f4a7c15ULL));}
static int addf(uint64_t fs[],int*n,uint64_t k){if(!k||*n>=MAXVF)return 0;for(int i=0;i<*n;i++)if(fs[i]==k)return 1;fs[(*n)++]=k;return 1;}

static int vfeatures(const VFrame*f,uint64_t fs[]){
    int xs[256],ys[256],cs[256],nn=0,n=0;
    for(int y=0;y<FH;y++)for(int x=0;x<FW;x++)if(f->p[y][x] && nn<256){xs[nn]=x;ys[nn]=y;cs[nn]=f->p[y][x];nn++;}
    /* D1: generic visible color relations.  Local world/world relations plus
       all relations involving a moving-color or visible reference pixel. */
    uint64_t anchor_rel[256][24]; int ar_n[256]; memset(ar_n,0,sizeof(ar_n));
    for(int i=0;i<nn;i++){
        for(int j=0;j<nn;j++)if(i!=j){
            int dx=xs[j]-xs[i],dy=ys[j]-ys[i];
            int local=(abs(dx)<=5 && abs(dy)<=5);
            int informative=(cs[i]==2 || cs[j]==2 || cs[i]==3 || cs[j]==3);
            if(!local && !informative)continue;
            uint64_t r=relkey(cs[i],cs[j],dx,dy); addf(fs,&n,r);
            if(ar_n[i]<24)anchor_rel[i][ar_n[i]++]=r;
        }
    }
    /* D2: ordinary collision of two relations sharing the same visible pixel. */
    for(int i=0;i<nn;i++){
        int m=ar_n[i];
        for(int a=0;a<m;a++)for(int b=a+1;b<m;b++){
            if(n>=MAXVF) break;
            addf(fs,&n,pairkey(anchor_rel[i][a],anchor_rel[i][b]));
        }
        if(n>=MAXVF)break;
    }
    return n;
}

static void op_init(OpField*f){memset(f,0,sizeof(*f));f->tab=calloc(OP_HT_SIZE,sizeof(OpEntry));if(!f->tab){fprintf(stderr,"op alloc\n");exit(2);}}
static void op_free(OpField*f){free(f->tab);}
static OpEntry*op_find(OpField*f,uint64_t k,int create){uint32_t h=(uint32_t)mix64(k)&OP_HT_MASK;for(uint32_t i=0;i<OP_HT_SIZE;i++){OpEntry*e=&f->tab[(h+i)&OP_HT_MASK];if(e->key==k)return e;if(!e->key){if(!create)return NULL;e->key=k;f->n++;return e;}}return NULL;}
static void op_learn(OpField*f,const VFrame*vf,int action){uint64_t fs[MAXVF];int nf=vfeatures(vf,fs);f->base[action]+=1;f->total+=1;for(int i=0;i<nf;i++){OpEntry*e=op_find(f,fs[i],1);if(!e)continue;e->c[action]+=1;e->total+=1;}}
static int op_predict(OpField*f,const VFrame*vf,double score[4]){
    uint64_t fs[MAXVF];int nf=vfeatures(vf,fs);for(int a=0;a<4;a++)score[a]=(f->base[a]+1.0)/(f->total+4.0)-0.25;
    double z[4]={0,0,0,0};
    for(int i=0;i<nf;i++){
        OpEntry*e=op_find(f,fs[i],0);if(!e||e->total<2)continue;double conf=e->total/(e->total+4.0);
        for(int a=0;a<4;a++){double p=(e->c[a]+1.0)/(e->total+4.0);double d=(p-0.25)*conf;score[a]+=d*fabs(d);z[a]+=fabs(d);}
    }
    int best=0;double bs=-1e99;for(int a=0;a<4;a++){double s=z[a]>0?score[a]/z[a]:score[a];if(s>bs){bs=s;best=a;}}
    return best; /* external measurement boundary */
}

static int min_active_x(const State*s){int m=s->w;for(int y=0;y<s->h;y++)for(int x=0;x<s->w;x++)if(s->a[y][x]&&x<m)m=x;return m;}
static int max_active_x(const State*s){int m=-1;for(int y=0;y<s->h;y++)for(int x=0;x<s->w;x++)if(s->a[y][x]&&x>m)m=x;return m;}
static int board_holes(const State*s){int holes=0;for(int x=0;x<s->w;x++){int seen=0;for(int y=0;y<s->h;y++){if(s->world[y][x])seen=1;else if(seen)holes++;}}return holes;}
static int col_height(const State*s,int x){for(int y=0;y<s->h;y++)if(s->world[y][x])return s->h-y;return 0;}
static double board_score(const State*s,int clears){double agg=0,bump=0,maxh=0;for(int x=0;x<s->w;x++){int h=col_height(s,x);agg+=h;if(h>maxh)maxh=h;if(x)bump+=fabs((double)h-col_height(s,x-1));}int holes=board_holes(s);return 0.55*agg+1.8*holes+0.35*bump+0.8*maxh-8.0*clears;}

static int apply_phys_once(const State*s,int op,State*out){const int amap[4]={0,1,2,3};unsigned long long save=truth_clear_events;reality(s,op,amap,out);int changed=!same(s,out);truth_clear_events=save;return changed;}
static int drop_one_piece(const State*s,State*out,int*clears){const int amap[4]={0,1,2,3};State x=*s,y;unsigned long long c0=truth_clear_events;for(int k=0;k<64;k++){reality(&x,2,amap,&y);x=y;if(last_truth_spawn_attempt){*out=x;*clears=(int)(truth_clear_events-c0);return 1;}}*out=x;*clears=(int)(truth_clear_events-c0);return 0;}

/* External demonstrator only.  It may use search/heuristic; BPC never sees
   these quantities, only the resulting visible action. */
static int teacher_action(const State*s){
    double best=1e99;int best_r=0,best_shift=0;int curx=min_active_x(s);
    for(int r=0;r<4;r++){
        State sr=*s,tmp;int valid=1;
        for(int k=0;k<r;k++){apply_phys_once(&sr,3,&tmp);if(same(&sr,&tmp)&&k==0){/* symmetric piece is still valid */}sr=tmp;}
        int mn=min_active_x(&sr),mx=max_active_x(&sr);if(mn<0||mx<0)continue;
        for(int target=0;target<=sr.w-1-(mx-mn);target++){
            State sh=sr;int delta=target-mn;int op=delta<0?0:1;for(int k=0;k<abs(delta);k++){apply_phys_once(&sh,op,&tmp);if(same(&sh,&tmp)){valid=0;break;}sh=tmp;}if(!valid){valid=1;continue;}
            State end;int clears=0;if(!drop_one_piece(&sh,&end,&clears))continue;double sc=board_score(&end,clears);
            if(sc<best){best=sc;best_r=r;best_shift=target-curx;}
        }
    }
    if(best_r>0)return 3;
    if(best_shift<0)return 0;
    if(best_shift>0)return 1;
    return 2;
}

static void game_apply(State*s,int a,long long*locks,long long*clears,long long*gameovers){const int amap[4]={0,1,2,3};State n;unsigned long long cb=truth_clear_events;reality(s,a,amap,&n);int attempt=last_truth_spawn_attempt,success=last_truth_spawn_success;*clears+=(long long)(truth_clear_events-cb);if(attempt)(*locks)++;*s=n;if(attempt&&success)refresh_preview(s);else if(attempt&&!success){(*gameovers)++;reset_game(s);}}

static void train_operation(OpField*f,int decisions){State s;reset_game(&s);long long L=0,C=0,G=0;for(int i=0;i<decisions;i++){int a=teacher_action(&s);VFrame vf;vrender(&s,&vf);op_learn(f,&vf,a);game_apply(&s,a,&L,&C,&G);}printf("STAGE_B train decisions=%d entries=%d teacher_locks=%lld clears=%lld gameovers=%lld\n",decisions,f->n,L,C,G);}
static void eval_operation(OpField*f,int decisions){State s;reset_game(&s);long long ok=0;for(int i=0;i<decisions;i++){int ta=teacher_action(&s);VFrame vf;double sc[4];vrender(&s,&vf);int pa=op_predict(f,&vf,sc);ok+=pa==ta;long long z=0;game_apply(&s,ta,&z,&z,&z);}printf("STAGE_B heldout_action_exact=%lld/%d (%.2f%%)\n",ok,decisions,100.0*ok/decisions);}
static void eval_autoplay(OpField*f,int ticks,const char*tag){State s;reset_game(&s);long long L=0,C=0,G=0,match=0;for(int i=0;i<ticks;i++){int ta=teacher_action(&s);VFrame vf;double sc[4];vrender(&s,&vf);int a=op_predict(f,&vf,sc);match+=a==ta;game_apply(&s,a,&L,&C,&G);}printf("STAGE_B %s ticks=%d teacher_match=%lld/%d locks=%lld clears=%lld gameovers=%lld\n",tag,ticks,match,ticks,L,C,G);}


static void eval_teacher_play(int ticks,const char*tag){State s;reset_game(&s);long long L=0,C=0,G=0;for(int i=0;i<ticks;i++){int a=teacher_action(&s);game_apply(&s,a,&L,&C,&G);}printf("TEACHER %s ticks=%d locks=%lld clears=%lld gameovers=%lld\n",tag,ticks,L,C,G);}

static void onpolicy_action_residual(OpField*f,int ticks,int round){
    State s;reset_game(&s);long long L=0,C=0,G=0,match=0;
    for(int i=0;i<ticks;i++){
        VFrame vf;double sc[4];vrender(&s,&vf);int pa=op_predict(f,&vf,sc);int ta=teacher_action(&s);
        match+=pa==ta;
        /* True action residual writeback: mature correct predictions receive no write. */
        if(pa!=ta) op_learn(f,&vf,ta);
        game_apply(&s,pa,&L,&C,&G);
    }
    printf("ACTION_RESIDUAL_QEWB round=%d visited=%d preupdate_match=%lld/%d locks=%lld clears=%lld gameovers=%lld entries=%d\n",round,ticks,match,ticks,L,C,G,f->n);
}
