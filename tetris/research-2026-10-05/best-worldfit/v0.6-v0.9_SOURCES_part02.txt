/* ---------- generic boundary-closure carrier; no row grouping ---------- */
typedef struct{uint32_t z[4],o[4];}ClosureGate;
static const int CDX[4]={1,-1,0,0},CDY[4]={0,0,1,-1};
static float clp(const ClosureGate*g,int d){uint32_t z=g->z[d],o=g->o[d];return((float)o+.5f)/((float)(z+o)+1.f);}
static void clobs(ClosureGate*g,int d,int yes){if(yes)g->o[d]++;else g->z[d]++;}
/* A candidate closure is purely physical: starting from an occupied information
   instance whose predecessor in direction d lies outside the lattice, propagate
   the same local direction through occupied neighbors. If the chain reaches the
   opposite boundary, all participating instances form the candidate seed. */
static int boundary_seed_dir(const uint8_t*b,int dx,int dy,uint8_t seed[N]){
  memset(seed,0,N);int any=0;
  for(int sy=0;sy<H;sy++)for(int sx=0;sx<W;sx++){
    if(!b[sy*W+sx])continue;
    int px=sx-dx,py=sy-dy;if(px>=0&&px<W&&py>=0&&py<H)continue;
    uint8_t tmp[N]={0};int x=sx,y=sy;
    while(x>=0&&x<W&&y>=0&&y<H&&b[y*W+x]){tmp[y*W+x]=1;x+=dx;y+=dy;}
    if(x<0||x>=W||y<0||y>=H){for(int i=0;i<N;i++)if(tmp[i])seed[i]=1;any=1;}
  }
  return any;
}
/* Local exclusion dynamics for multiple hole carriers. Content transfer is not
   a LineClear-specific transform: it uses the already learned one-step DOWN. */
static void apply_hole_seed(const Move*mv,uint8_t*b,const uint8_t seed[N]){
  uint8_t hole[N];memcpy(hole,seed,N);
  for(int i=0;i<N;i++)if(hole[i])b[i]=0;
  for(int step=0;step<N;step++){
    uint8_t movehole[N]={0},next_hole[N],nb[N];int live=0;float moved[N];
    shifted_probs(mv,b,2,moved);memcpy(next_hole,hole,N);memcpy(nb,b,N);
    for(int x=0;x<W;x++)if(hole[x]){next_hole[x]=0;nb[x]=0;}
    for(int y=1;y<H;y++)for(int x=0;x<W;x++){int i=y*W+x,up=(y-1)*W+x;movehole[i]=(uint8_t)(hole[i]&&!hole[up]);}
    for(int y=1;y<H;y++)for(int x=0;x<W;x++){int i=y*W+x,up=(y-1)*W+x;if(movehole[i]){nb[i]=(uint8_t)(moved[i]>.5f);nb[up]=0;next_hole[i]=0;next_hole[up]=1;}}
    memcpy(b,nb,N);memcpy(hole,next_hole,N);for(int i=0;i<N;i++)live|=hole[i];if(!live)break;
  }
}
static int hdist(const uint8_t*a,const uint8_t*b){int n=0;for(int i=0;i<N;i++)n+=a[i]!=b[i];return n;}
static int clear_all_model(const ClosureGate*g,const Move*mv,uint8_t*b){
  uint8_t seed[N]={0};int any=0;
  for(int d=0;d<4;d++)if(clp(g,d)>.5f){uint8_t q[N];if(boundary_seed_dir(b,CDX[d],CDY[d],q)){for(int i=0;i<N;i++)seed[i]|=q[i];any=1;}}
  if(!any)return 0;
  apply_hole_seed(mv,b,seed);
  return 1;
}

typedef struct{float w[ACTS][2];} ActionLife;
typedef struct{Move mv;MoveBlock mb;RotRel rr;Couple cp;SpawnRel sp;ClosureGate cl;ActionLife life;}Model;
static void model_init(Model*m){memset(m,0,sizeof(*m));train_relation_rot(&m->rr,60000);train_move(&m->mv,30);train_mblock(&m->mv,&m->mb,100);train_couple(&m->cp,300);train_spawn_rel(&m->sp,1000);train_closure_gate(&m->cl,&m->mv,300);}
static void learned_spawn(const Model*m,const uint8_t prev[PREV],const uint8_t*world,uint8_t*a,float g,uint8_t*gameover){
    float field[N];spawn_rel_field(&m->sp,prev,field);
    float C=spawn_rel_carrier(&m->sp,prev,world),gs,gg;cg(&m->cp,C,&gs,&gg);
    float spawn_gain=g*gs;*gameover=(uint8_t)(g*gg>.5f);
    for(int i=0;i<N;i++)if(spawn_gain*field[i]>.5f)a[i]=1;
}
/* v1.7: after the external one-step relation, downstream learned relations are
   repeatedly allowed to act until the field is quiescent.  No explicit
   Lock->Clear->Spawn call chain is used. */
static float model_down_once(const Model*m,State*s){
    if(s->gameover)return 0.f;
    float C=move_carrier(&m->mv,&m->mb,s->active,s->world,2),gm,gl;
    cg(&m->cp,C,&gm,&gl);
    float ma[N];shifted_probs(&m->mv,s->active,2,ma);
    uint8_t na[N]={0};
    for(int i=0;i<N;i++){
      na[i]=(uint8_t)(gm*ma[i]>.5f);
      
      if(gl*(float)s->active[i]>.5f)s->world[i]=1;
    }
    memcpy(s->active,na,N);
    float pending=gl;
    /* Pure relaxation: no quiescence gate and no fixed Lock->Clear->Spawn schedule.
       Spawn and closure are ordinary learned consequences that remain eligible on
       every microstep; the state is read only after the joint field reaches a fixed point. */
    for(int micro=0;micro<H+8;micro++){
      uint8_t beforeA[N],beforeW[N],beforeG=s->gameover;
      memcpy(beforeA,s->active,N);memcpy(beforeW,s->world,N);
      learned_spawn(m,s->preview,s->world,s->active,pending,&s->gameover);
      (void)clear_all_model(&m->cl,&m->mv,s->world);
      int changed=!eq(beforeA,s->active,N)||!eq(beforeW,s->world,N)||beforeG!=s->gameover;
      if(!changed)break;
    }
    return gl;
}
static void model_move_lr(const Model*m,State*s,int dir){float C=move_carrier(&m->mv,&m->mb,s->active,s->world,dir);float ma[N];shifted_probs(&m->mv,s->active,dir,ma);for(int i=0;i<N;i++)s->active[i]=(uint8_t)((C*ma[i]+(1-C)*s->active[i])>.5f);}
static void model_rot(const Model*m,State*s){uint8_t a[N];relation_rot_predict(&m->rr,s->active,s->world,a);memcpy(s->active,a,N);}
static float rot_carrier_rel(const Model*m,const State*s){int sx=rr_phase(m->rr.phase[0]),sy=rr_phase(m->rr.phase[1]);uint8_t q[N];if(!sx||!sy||!rr_reconstruct(&m->rr,s->active,sx,sy,q))return 0.f;for(int i=0;i<N;i++)if(q[i]&&s->world[i])return 0.f;return 1.f;}
static float action_carrier(const Model*m,const State*s,int act){if(act==ACT_LEFT)return move_carrier(&m->mv,&m->mb,s->active,s->world,0);if(act==ACT_RIGHT)return move_carrier(&m->mv,&m->mb,s->active,s->world,1);if(act==ACT_DOWN||act==ACT_DROP)return move_carrier(&m->mv,&m->mb,s->active,s->world,2);if(act==ACT_ROT)return rot_carrier_rel(m,s);return 0.f;}

static void audit_rollout(const Model*m,int episodes,int maxsteps,uint64_t seed){
