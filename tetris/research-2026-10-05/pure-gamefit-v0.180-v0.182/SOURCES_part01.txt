static uint64_t mix64(uint64_t x){x^=x>>33;x*=0xff51afd7ed558ccdULL;x^=x>>33;x*=0xc4ceb9fe1a85ec53ULL;x^=x>>33;return x;}
static uint64_t prim(int c1,int c2,int dx,int dy,int dt){uint64_t u=1ULL<<63;u|=(uint64_t)(c1&7)<<56;u|=(uint64_t)(c2&7)<<52;u|=(uint64_t)((dx+8)&15)<<44;u|=(uint64_t)((dy+8)&15)<<36;u|=(uint64_t)(dt&3)<<34;return u;}

static void upd_one(Field*f,uint64_t iw,uint64_t ow,double wt,int yes){if(wt<=0)return;Coupling*c=cfind(f,iw,ow,1);if(!c)return;c->total+=wt;c->yes+=wt*(!!yes);}
static void upd_value(Field*f,uint64_t iw,uint64_t ow,double wt,double target){if(wt<=0)return;if(target<0)target=0;if(target>1)target=1;Coupling*c=cfind(f,iw,ow,1);if(!c)return;c->total+=wt;c->yes+=wt*target;}
static void upd(Field*f,int act,uint64_t iw,uint64_t ow,double wt,int yes){
    uint64_t aiw=collide(ACTION_WAVE(act),iw);Coupling*g=cfind(f,iw,ow,0),*ac0=cfind(f,aiw,ow,0);
    double pg=(g&&g->total>0)?cp(g):0.5;double pa=(ac0&&ac0->total>0)?cp(ac0):pg;double pred=pg+2.0*(pa-0.5);if(pred>1)pred=1;if(pred<0)pred=0;
    double t=yes?1.0:0.0,r=t-pg;double parent_w=wt,child_w=0.0;
    if(residual_context_n>0){int contrad=((pred>=0.5)!=(!!yes));if(contrad){double ev=(g?g->total:0)+(ac0?ac0->total:0);double inertia=ev/(ev+1.0);if(inertia<0)inertia=0;if(inertia>1)inertia=1;parent_w=wt*(1.0-inertia);child_w=wt*inertia;}}
    upd_one(f,iw,ow,parent_w,yes);upd_value(f,aiw,ow,parent_w,0.5+0.5*r);
    if(child_w>0&&residual_context_n>0){double each=child_w/(double)residual_context_n;for(int rc=0;rc<residual_context_n;rc++){uint64_t rw=residual_context_waves[rc],ciw=(rw<iw)?collide(rw,iw):collide(iw,rw),caiw=collide(ACTION_WAVE(act),ciw);upd_one(f,ciw,ow,each,yes);upd_value(f,caiw,ow,each,yes?1.0:0.0);}}
}

static uint64_t PLANE(int ch){return prim(ch+1,ch+1,0,0,0);}
static uint64_t REL(int ch,int dx,int dy){(void)ch;return prim(3,4,dx,dy,0);}

static int is_xrel(uint64_t w,int sc,int tc,int*dx,int*dy){if((w>>63)==0)return 0;int c1,c2,dt;unprim(w,&c1,&c2,dx,dy,&dt);return c1==sc+1&&c2==tc+1&&dt==1;}
static uint64_t OVERFLOW_BASE(void){return collide(prim(0,0,0,0,0),prim(0,0,0,0,1));}

/* No task-specific span token. A closure context is the ordinary collision of
   the universal unclosed/overflow wave with the local relation direction. */
static uint64_t CLOSURE_WAVE_DIR(int ch,int dx,int dy){
    (void)ch;
    uint64_t r1=REL(0,dx,dy),r2=REL(0,-dx,-dy),pair;
    if(r1>r2){uint64_t t=r1;r1=r2;r2=t;}pair=collide(r1,r2);
    uint64_t a=OVERFLOW_BASE(),b=pair;if(a>b){uint64_t t=a;a=b;b=t;}return collide(a,b);
}
/* Closure directions are not pre-enumerated. They are the unit relation waves
   physically present in the current state. */
static int state_unit_dirs(const State*s,int ch,int dx[],int dy[],int cap){
    int n=0;
    for(int y=0;y<s->h;y++)for(int x=0;x<s->w;x++)if(pget(s,ch,x,y)){
        for(int oy=-1;oy<=1;oy++)for(int ox=-1;ox<=1;ox++){
            if(abs(ox)+abs(oy)!=1)continue;
            int nx=x+ox,ny=y+oy;if(!in(s,nx,ny)||!pget(s,ch,nx,ny))continue;
            int seen=0;for(int i=0;i<n;i++)seen|=(dx[i]==ox&&dy[i]==oy);
            if(!seen&&n<cap){dx[n]=ox;dy[n]=oy;n++;}
        }
    }
    return n;
}
/* Direction-symmetric physical boundary carry.  Opposite directions generate
   the same closure-wave identity; readout deduplicates that wave identity. */
static int boundary_carry_seed_dir(const State*s,int ch,int dx,int dy,uint8_t seed[MAXH][MAXW]){
    uint8_t reach[MAXH][MAXW],next[MAXH][MAXW],ack[MAXH][MAXW];
    memset(reach,0,sizeof(reach));memset(seed,0,MAXH*MAXW*sizeof(uint8_t));
    /* All source-boundary occupied voxels emit the same binary carry wave. */
    for(int y=0;y<s->h;y++)for(int x=0;x<s->w;x++)
        if(pget(s,ch,x,y)&&!in(s,x-dx,y-dy))reach[y][x]=1;
    for(int st=0;st<s->w+s->h;st++){
        memcpy(next,reach,sizeof(next));int changed=0;
        for(int y=0;y<s->h;y++)for(int x=0;x<s->w;x++)if(reach[y][x]){
            int nx=x+dx,ny=y+dy;
            if(in(s,nx,ny)&&pget(s,ch,nx,ny)&&!next[ny][nx]){next[ny][nx]=1;changed=1;}
        }
        memcpy(reach,next,sizeof(reach));if(!changed)break;
    }
    memset(ack,0,sizeof(ack));int closed=0;
    for(int y=0;y<s->h;y++)for(int x=0;x<s->w;x++)
        if(reach[y][x]&&!in(s,x+dx,y+dy)){ack[y][x]=1;closed=1;}
    if(!closed)return 0;
    /* Opposite-boundary echo propagates back only through the forward wave.
       No endpoint selection, row index, path list, or visited traversal. */
    for(int st=0;st<s->w+s->h;st++){
        memcpy(next,ack,sizeof(next));int changed=0;
        for(int y=0;y<s->h;y++)for(int x=0;x<s->w;x++)if(ack[y][x]){
            int px=x-dx,py=y-dy;
            if(in(s,px,py)&&reach[py][px]&&pget(s,ch,px,py)&&!next[py][px]){next[py][px]=1;changed=1;}
        }
        memcpy(ack,next,sizeof(ack));if(!changed)break;
    }
    memcpy(seed,ack,MAXH*MAXW*sizeof(uint8_t));return 1;
}
static int boundary_carry_closed_dir(const State*s,int ch,int dx,int dy){uint8_t seed[MAXH][MAXW];return boundary_carry_seed_dir(s,ch,dx,dy,seed);}

typedef struct { uint64_t w; int dx,dy; } LocalRel;
static int add_local_rel(LocalRel r[],int*n,int cap,int ch,int dx,int dy){
    if(abs(dx)>1||abs(dy)>1)return 0;
    uint64_t w=REL(ch,dx,dy);
    for(int i=0;i<*n;i++)if(r[i].w==w)return 1;
    if(*n>=cap)return 0;
    r[*n]=(LocalRel){w,dx,dy};(*n)++;return 1;
}
static int learned_closure_relations(Field*f,int act,int ch,uint64_t iw,LocalRel r[],int cap){
    int n=0;uint64_t aiw=collide(ACTION_WAVE(act),iw);
    for(int i=0;i<f->n&&n<cap;i++)if(f->c[i].inw==iw||f->c[i].inw==aiw){
        int c1,c2,dx,dy,dt;unprim(f->c[i].outw,&c1,&c2,&dx,&dy,&dt);
        if(c1!=3||c2!=4||dt!=0||abs(dx)>1||abs(dy)>1)continue;
        add_local_rel(r,&n,cap,ch,dx,dy);
    }
    return n;
}

/* Generic residual-to-relation birth.  This learner knows no row/span/closure
   geometry.  Given any already-grounded residual context wave, it asks which
   local address displacement best explains the pixels that actually changed
   between the model's intermediate prediction and reality. */
static int residual_relation_candidates(const State*mid,const State*truth,int ch,LocalRel r[],int cap){
    int n=0;
    for(int y=0;y<mid->h;y++)for(int x=0;x<mid->w;x++){
        int mv=pget(mid,ch,x,y),tv=pget(truth,ch,x,y);if(mv==tv||!tv)continue;
        for(int sy=y-1;sy<=y+1;sy++)for(int sx=x-1;sx<=x+1;sx++){
            if(!in(mid,sx,sy)||!pget(mid,ch,sx,sy))continue;
            add_local_rel(r,&n,cap,ch,x-sx,y-sy);
        }
    }
    return n;
}
static void learn_generic_residual_relations(Field*f,const State*mid,const State*truth,int act,int ch,uint64_t iw){
    LocalRel cand[16];int nc=residual_relation_candidates(mid,truth,ch,cand,16);if(nc<=0)return;
    int nr=0;for(int y=0;y<mid->h;y++)for(int x=0;x<mid->w;x++)nr+=pget(mid,ch,x,y)!=pget(truth,ch,x,y);if(nr<=0)return;
    for(int j=0;j<nc;j++){
        double sc=0;
        for(int y=0;y<mid->h;y++)for(int x=0;x<mid->w;x++)if(pget(mid,ch,x,y)!=pget(truth,ch,x,y)){
            int sx=x-cand[j].dx,sy=y-cand[j].dy;int pred=(in(mid,sx,sy)?pget(mid,ch,sx,sy):0);
            sc+=(pred==pget(truth,ch,x,y));
        }
        double target=sc/(double)nr;
        upd_value(f,iw,cand[j].w,2.0,target);
        upd_value(f,collide(ACTION_WAVE(act),iw),cand[j].w,4.0,target);
    }
}
static int mask_any(const State*s,uint8_t m[MAXH][MAXW]){for(int y=0;y<s->h;y++)for(int x=0;x<s->w;x++)if(m[y][x])return 1;return 0;}
static int closure_relax_once(Field*f,const State*s,int act,State*out){
    *out=*s;int changed=0;
    for(int ch=0;ch<PLANES;ch++){
        int cdxv[8],cdyv[8],nd=state_unit_dirs(s,ch,cdxv,cdyv,8);uint64_t seen[8]={0};int nseen=0;
        for(int d=0;d<nd;d++){
            int cdx=cdxv[d],cdy=cdyv[d];uint8_t seed[MAXH][MAXW];if(!boundary_carry_seed_dir(s,ch,cdx,cdy,seed))continue;
            uint64_t iw=CLOSURE_WAVE_DIR(ch,cdx,cdy);int dup=0;for(int j=0;j<nseen;j++)dup|=(seen[j]==iw);if(dup)continue;if(nseen<8)seen[nseen++]=iw;
            LocalRel cand[16];int nc=learned_closure_relations(f,act,ch,iw,cand,16);if(nc<=0)continue;
            double sx=0,sy=0,z=0;for(int j=0;j<nc;j++){double w=posw(f,act,iw,cand[j].w);sx+=w*cand[j].dx;sy+=w*cand[j].dy;z+=w;}
            if(z<=0)continue;
            double mag=hypot(sx,sy);
            if(mag<=1e-12)continue;
            int dx=(int)lround(sx/mag),dy=(int)lround(sy/mag);
            /* No Manhattan-only child gate. Any reality-born local relation in
               the radius-1 physical neighborhood may act. (0,0) is the natural
               identity and therefore has no state-changing relaxation.) */
            if(dx==0&&dy==0)continue;
            if(abs(dx)>1||abs(dy)>1)continue;
            uint8_t tmp[MAXH][MAXW],holes[MAXH][MAXW],next[MAXH][MAXW];memset(tmp,0,sizeof(tmp));memcpy(holes,seed,sizeof(holes));
            for(int y=0;y<s->h;y++)for(int x=0;x<s->w;x++)tmp[y][x]=pget(s,ch,x,y);
            while(mask_any(s,holes)){
                uint8_t nh[MAXH][MAXW],nb[MAXH][MAXW];memcpy(nb,tmp,sizeof(nb));memset(nh,0,sizeof(nh));int any=0;
                for(int y=0;y<s->h;y++)for(int x=0;x<s->w;x++)if(holes[y][x]){
                    int px=x-dx,py=y-dy;int v=0;
                    if(in(s,px,py)){
                        if(holes[py][px]){v=0;nh[y][x]=1;any=1;}
                        else{v=tmp[py][px];nh[py][px]=1;any=1;}
                    }
                    nb[y][x]=(uint8_t)v;
                }
                memcpy(tmp,nb,sizeof(tmp));memcpy(next,nh,sizeof(next));memcpy(holes,next,sizeof(holes));if(!any)break;
            }
            for(int y=0;y<s->h;y++)for(int x=0;x<s->w;x++){uint8_t old=pget(out,ch,x,y);pset(out,ch,x,y,tmp[y][x]);changed|=(old!=tmp[y][x]);}
        }
    }
    return changed;
}
static void closure_relax(Field*f,const State*s,int act,State*out){State a=*s,b;int safety=s->w*s->h;for(int k=0;k<safety;k++){if(!closure_relax_once(f,&a,act,&b)){*out=a;return;}a=b;}*out=a;}

static int is_shift_rel(uint64_t w,int*dx,int*dy){if((w>>63)==0)return 0;int c1,c2,dt;unprim(w,&c1,&c2,dx,dy,&dt);return c1==3&&c2==4&&dt==0;}
static int learned_shift(Field*f,int a,int ch,int*dx,int*dy,Mode m){
    int aa=m==FIELD_SHIFT?(a+1)%ACTS:a;uint64_t base=PLANE(ch),actin=collide(ACTION_WAVE(aa),base),outs[64];int no=0;
    for(int i=0;i<f->n&&no<64;i++)if(f->c[i].inw==base||f->c[i].inw==actin){int x,y;if(!is_shift_rel(f->c[i].outw,&x,&y))continue;int seen=0;for(int j=0;j<no;j++)seen|=outs[j]==f->c[i].outw;if(!seen)outs[no++]=f->c[i].outw;}
    double sx=0,sy=0,z=0;for(int i=0;i<no;i++){int x=0,y=0;if(!is_shift_rel(outs[i],&x,&y))continue;double w=posw(f,aa,base,outs[i]);sx+=w*x;sy+=w*y;z+=w;}
    if(z<=0)return 0;
    *dx=(int)lround(sx/z);*dy=(int)lround(sy/z);return 1;
}
static void learn_shift_output(Field*f,int a,int ch,int gx,int gy,double wt){
    uint64_t base=PLANE(ch),actin=collide(ACTION_WAVE(a),base),actual=REL(ch,gx,gy),outs[64];int no=0;
    for(int i=0;i<f->n&&no<63;i++)if(f->c[i].inw==base||f->c[i].inw==actin){int x,y;if(!is_shift_rel(f->c[i].outw,&x,&y))continue;int seen=0;for(int j=0;j<no;j++)seen|=outs[j]==f->c[i].outw;if(!seen)outs[no++]=f->c[i].outw;}
    int have=0;for(int i=0;i<no;i++)have|=outs[i]==actual;if(!have)outs[no++]=actual;
    for(int i=0;i<no;i++)upd(f,a,base,outs[i],wt,outs[i]==actual);
}
/* Relation-pair waves are reversible physical identities, not enumerated roles.
   Only one-hop relations that actually coexist in a current local voxel state
   can create a pair wave. */
static int local_relation_ok(int dx,int dy){return abs(dx)+abs(dy)==1;}
static unsigned relation_key(int dx,int dy){return (unsigned)(((dx+8)&15)<<4)|((unsigned)((dy+8)&15));}
static uint64_t pair_wave(int dx1,int dy1,int dx2,int dy2){
    if(!local_relation_ok(dx1,dy1)||!local_relation_ok(dx2,dy2)||(dx1==dx2&&dy1==dy2))return 0;
    unsigned a=relation_key(dx1,dy1),b=relation_key(dx2,dy2);
    if(a>b){int tx=dx1,ty=dy1;dx1=dx2;dy1=dy2;dx2=tx;dy2=ty;}
    return (3ULL<<62)|((uint64_t)((dx1+8)&15)<<12)|((uint64_t)((dy1+8)&15)<<8)|((uint64_t)((dx2+8)&15)<<4)|(uint64_t)((dy2+8)&15);
}
static int decode_pair_wave(uint64_t w,int*dx1,int*dy1,int*dx2,int*dy2){
    if((w>>62)!=3)return 0;
    *dx1=(int)((w>>12)&15)-8;*dy1=(int)((w>>8)&15)-8;
    *dx2=(int)((w>>4)&15)-8;*dy2=(int)(w&15)-8;
    return local_relation_ok(*dx1,*dy1)&&local_relation_ok(*dx2,*dy2)&&!(*dx1==*dx2&&*dy1==*dy2);
}
static int add_u64(uint64_t a[],int*n,int cap,uint64_t w){if(!w)return 0;for(int i=0;i<*n;i++)if(a[i]==w)return 1;if(*n>=cap)return 0;a[(*n)++]=w;return 1;}
static int local_pair_waves(const State*s,int ch,uint64_t out[],int cap){int n=0;for(int y=0;y<s->h;y++)for(int x=0;x<s->w;x++)if(pget(s,ch,x,y)){int dx[8],dy[8],nd=0;for(int oy=-1;oy<=1;oy++)for(int ox=-1;ox<=1;ox++){if(abs(ox)+abs(oy)!=1)continue;int nx=x+ox,ny=y+oy;if(in(s,nx,ny)&&pget(s,ch,nx,ny)){dx[nd]=ox;dy[nd]=oy;nd++;}}for(int i=0;i<nd;i++)for(int j=i+1;j<nd;j++)add_u64(out,&n,cap,pair_wave(dx[i],dy[i],dx[j],dy[j]));}return n;}
static int has_u64(const uint64_t a[],int n,uint64_t w){for(int i=0;i<n;i++)if(a[i]==w)return 1;return 0;}
static void learn_pair_outputs(Field*f,int act,uint64_t iw,const uint64_t actual[],int na,double wt){uint64_t aiw=collide(ACTION_WAVE(act),iw),cand[64];int nc=0;for(int i=0;i<f->n&&nc<64;i++)if(f->c[i].inw==iw||f->c[i].inw==aiw){int ax,ay,bx,by;if(decode_pair_wave(f->c[i].outw,&ax,&ay,&bx,&by))add_u64(cand,&nc,64,f->c[i].outw);}for(int i=0;i<na;i++)add_u64(cand,&nc,64,actual[i]);for(int i=0;i<nc;i++)upd(f,act,iw,cand[i],wt,has_u64(actual,na,cand[i]));}
static int current_unit_dirs(const State*s,int ch,int dx[],int dy[],int cap){return state_unit_dirs(s,ch,dx,dy,cap);}
static int add_dir_score(int dx[],int dy[],double sc[],int*n,int cap,int x,int y,double w){for(int i=0;i<*n;i++)if(dx[i]==x&&dy[i]==y){sc[i]+=w;return 1;}if(*n>=cap)return 0;dx[*n]=x;dy[*n]=y;sc[*n]=w;(*n)++;return 1;}
static int mapped_vec_local(Field*f,int a,int rdx,int rdy,double*outx,double*outy,Mode m){
    if(m==RELATION_OFF)return 0;
    int aa=m==FIELD_SHIFT?(a+1)%ACTS:a,sdx[16],sdy[16],ns=0,evidence=0;double score[16]={0};
    /* Latent context vocabulary comes only from relation pairs that have actually
       existed in past reality.  No fixed direction table and no whole-field
       candidate sweep. */
    int qdx[16],qdy[16],nq=0;
    for(int i=0;i<f->n;i++){
        int ax,ay,bx,by;if(!decode_pair_wave(f->c[i].inw,&ax,&ay,&bx,&by))continue;
        int x=99,y=99;if(ax==rdx&&ay==rdy){x=bx;y=by;}else if(bx==rdx&&by==rdy){x=ax;y=ay;}else continue;
        int seen=0;for(int j=0;j<nq;j++)seen|=(qdx[j]==x&&qdy[j]==y);if(!seen&&nq<16){qdx[nq]=x;qdy[nq]=y;nq++;}
    }
    for(int q=0;q<nq;q++){
        uint64_t iw=pair_wave(rdx,rdy,qdx[q],qdy[q]);if(!iw)continue;uint64_t aiw=collide(ACTION_WAVE(aa),iw),outs[64];int no=0;
        for(int i=0;i<f->n&&no<64;i++)if(f->c[i].inw==iw||f->c[i].inw==aiw){int ax,ay,bx,by;if(!decode_pair_wave(f->c[i].outw,&ax,&ay,&bx,&by))continue;add_u64(outs,&no,64,f->c[i].outw);}
        double localz=0;int ldx[16],ldy[16],nl=0;double ls[16]={0};
        for(int i=0;i<no;i++){int ax=0,ay=0,bx=0,by=0;if(!decode_pair_wave(outs[i],&ax,&ay,&bx,&by))continue;double w=posw(f,aa,iw,outs[i]);if(w<=0)continue;add_dir_score(ldx,ldy,ls,&nl,16,ax,ay,w);add_dir_score(ldx,ldy,ls,&nl,16,bx,by,w);localz+=w;}
        if(localz>0){for(int i=0;i<nl;i++)add_dir_score(sdx,sdy,score,&ns,16,ldx[i],ldy[i],ls[i]/localz);evidence++;}
    }
    if(!evidence||ns<=0)return 0;
    double sx=0,sy=0;for(int i=0;i<ns;i++){sx+=score[i]*sdx[i];sy+=score[i]*sdy[i];}double mag=hypot(sx,sy);if(mag<=1e-9)return 0;*outx=sx/mag;*outy=sy/mag;return 1;
}
static void address_anchor(const State*s,int ch,int*ax,int*ay){uint8_t v[MAXH][MAXW],h[MAXH][MAXW];memset(v,0,sizeof(v));memset(h,0,sizeof(h));for(int y=0;y<s->h;y++)for(int x=0;x<s->w;x++)v[y][x]=h[y][x]=!!pget(s,ch,x,y);for(int st=0;st<s->h;st++)for(int y=1;y<s->h;y++)for(int x=0;x<s->w;x++)v[y-1][x]|=v[y][x];for(int st=0;st<s->w;st++)for(int x=1;x<s->w;x++)for(int y=0;y<s->h;y++)h[y][x-1]|=h[y][x];int x=0;while(x<s->w&&!v[0][x])x++;int y=0;while(y<s->h&&!h[y][0])y++;*ax=x;*ay=y;}
static int learned_targets(Field*f,const State*s,int a,int ch,int tx[],int ty[],Mode m){
    int x[128],y[128],n=collectp(s,ch,x,y,128);if(n<=0||n>128)return -1;
    int gx,gy,anchorx,anchory;address_anchor(s,ch,&anchorx,&anchory);if(!learned_shift(f,a,ch,&gx,&gy,m))return -1;
    int rdx[8],rdy[8],nd=current_unit_dirs(s,ch,rdx,rdy,8);double mdx[8],mdy[8];
    for(int r=0;r<nd;r++)if(!mapped_vec_local(f,a,rdx[r],rdy[r],&mdx[r],&mdy[r],m))return -1;
    int idx[MAXH][MAXW];for(int yy=0;yy<MAXH;yy++)for(int xx=0;xx<MAXW;xx++)idx[yy][xx]=-1;for(int i=0;i<n;i++)idx[y[i]][x[i]]=i;
    double ox[128]={0},oy[128]={0},nxv[128],nyv[128];
    int relax_safety=s->w*s->h*8;for(int pass=0;pass<relax_safety;pass++){
        double maxd=0;
        for(int i=0;i<n;i++){
            double sx=0,sy=0,z=0;
            for(int r=0;r<nd;r++){
                int xx=x[i]+rdx[r],yy=y[i]+rdy[r];if(!in(s,xx,yy))continue;int j=idx[yy][xx];if(j<0)continue;
                sx+=ox[j]-mdx[r];sy+=oy[j]-mdy[r];z+=1.0;
            }
            if(z>0){double tx0=sx/z,ty0=sy/z;nxv[i]=0.5*ox[i]+0.5*tx0;nyv[i]=0.5*oy[i]+0.5*ty0;}else{nxv[i]=ox[i];nyv[i]=oy[i];}
        }
        double mx=0,my=0;for(int i=0;i<n;i++){mx+=nxv[i];my+=nyv[i];}mx/=n;my/=n;
        for(int i=0;i<n;i++){nxv[i]-=mx;nyv[i]-=my;double d=fabs(nxv[i]-ox[i])+fabs(nyv[i]-oy[i]);if(d>maxd)maxd=d;ox[i]=nxv[i];oy[i]=nyv[i];}
        if(maxd<1e-8)break;
    }
    double mnx=ox[0],mny=oy[0];for(int i=1;i<n;i++){if(ox[i]<mnx)mnx=ox[i];if(oy[i]<mny)mny=oy[i];}
    for(int i=0;i<n;i++){tx[i]=anchorx+gx+(int)lround(ox[i]-mnx);ty[i]=anchory+gy+(int)lround(oy[i]-mny);}return n;
}
typedef struct { uint8_t b[16]; } MassWave;
static void mass_inject_one(MassWave*m){int carry=1;for(int i=0;i<16&&carry;i++){int old=m->b[i];m->b[i]=(uint8_t)(old^carry);carry=old&carry;}}
static MassWave information_mass_wave(const State*s,int ch){MassWave m;memset(&m,0,sizeof(m));for(int y=0;y<s->h;y++)for(int x=0;x<s->w;x++)if(pget(s,ch,x,y))mass_inject_one(&m);return m;}
static uint64_t mass_state_wave(const State*s,int ch){
    MassWave m=information_mass_wave(s,ch);uint64_t w=PLANE(ch);
    for(int i=0;i<16;i++)w=collide(w,prim(5,5,i,m.b[i]?1:0,3));
    return w;
}
static uint64_t cross_mass_context(const State*s,int tc,int sc){
    uint64_t a=mass_state_wave(s,tc),b=PLANE(sc);if(a>b){uint64_t t=a;a=b;b=t;}return collide(a,b);
}
static int mass_wave_closed(const State*a,const State*b,int ch){MassWave x=information_mass_wave(a,ch),y=information_mass_wave(b,ch);for(int i=0;i<16;i++)if(x.b[i]!=y.b[i])return 0;return 1;}
static int mass_wave_equal_cross(const State*a,int ca,const State*b,int cb){MassWave x=information_mass_wave(a,ca),y=information_mass_wave(b,cb);for(int i=0;i<16;i++)if(x.b[i]!=y.b[i])return 0;return 1;}
static int all_mass_waves_closed(const State*a,const State*b){
    for(int ch=0;ch<PLANES;ch++)if(!mass_wave_closed(a,b,ch))return 0;
    return 1;
}
static double cross_relation_coherence(const State*src,int sc,const State*dst,int tc){
    uint64_t a[32],b[32];int na=local_pair_waves(src,sc,a,32),nb=local_pair_waves(dst,tc,b,32);
    if(na<=0||nb<=0)return 0.0;
    int hit=0;for(int i=0;i<na;i++)hit+=has_u64(b,nb,a[i]);
    int uni=na+nb-hit;if(uni<=0)return 0.0;
    return (double)hit/(double)uni;
}
static void learn_mass_context_birth(Field*f,const State*mid,const State*truth,int act){
    for(int tc=0;tc<PLANES;tc++){
        /* No empty-target gate.  A target plane participates whenever its
           information-mass wave actually changed in reality.  Zero mass is
           merely one ordinary state of the same mass-wave carrier. */
        if(mass_wave_closed(mid,truth,tc))continue;
        for(int sc=0;sc<PLANES;sc++)if(sc!=tc){
            if(!mass_wave_equal_cross(mid,sc,truth,tc))continue;
            double coh=cross_relation_coherence(mid,sc,truth,tc);if(coh<=0.0)continue;
            int sx,sy,tx,ty;address_anchor(mid,sc,&sx,&sy);address_anchor(truth,tc,&tx,&ty);int dx=tx-sx,dy=ty-sy;
            if(abs(dx)>7||abs(dy)>7)continue;
            uint64_t iw=cross_mass_context(mid,tc,sc),actual=XREL(sc,tc,dx,dy);
            upd(f,act,iw,actual,coh,1);
        }
    }
}
static void mass_context_birth_relax(Field*f,const State*s,int act,State*out){
    *out=*s;
    for(int tc=0;tc<PLANES;tc++){
        double score[MAXH][MAXW];memset(score,0,sizeof(score));
        for(int sc=0;sc<PLANES;sc++)if(sc!=tc){
            uint64_t iw=cross_mass_context(s,tc,sc),aiw=collide(ACTION_WAVE(act),iw),outs[64];int no=0;
            for(int i=0;i<f->n&&no<64;i++)if(f->c[i].inw==iw||f->c[i].inw==aiw){int dx,dy;if(!is_xrel(f->c[i].outw,sc,tc,&dx,&dy))continue;int seen=0;for(int j=0;j<no;j++)seen|=outs[j]==f->c[i].outw;if(!seen)outs[no++]=f->c[i].outw;}
            for(int oi=0;oi<no;oi++){
                int dx=0,dy=0;if(!is_xrel(outs[oi],sc,tc,&dx,&dy))continue;
                /* Every D2 relation wave participates continuously.  No
                   internal birth threshold and no winner direction.  Its
                   positive amplitude is converted to physical contribution;
                   discreteness happens only when target voxels are measured. */
                double amp=2.0*posw(f,act,iw,outs[oi]);if(amp<=0)continue;
                uint8_t carry[MAXH][MAXW];memset(carry,0,sizeof(carry));
                for(int y=0;y<s->h;y++)for(int x=0;x<s->w;x++)if(pget(s,sc,x,y)){
                    int nx=x+dx,ny=y+dy,ib=0;int mask=rawmask(s,nx,ny,&ib);
                    if(bit(f,act,CTX_MASK(tc,ib,mask),OVERFLOW(tc)))carry[y][x]=1;
                }
                int rdx[8],rdy[8],nd=state_unit_dirs(s,sc,rdx,rdy,8);
                for(int pass=0;pass<s->w*s->h;pass++){
                    uint8_t next[MAXH][MAXW];memcpy(next,carry,sizeof(next));int changed=0;
                    for(int y=0;y<s->h;y++)for(int x=0;x<s->w;x++)if(pget(s,sc,x,y)&&carry[y][x]){
                        for(int r=0;r<nd;r++){int qx=x+rdx[r],qy=y+rdy[r];if(in(s,qx,qy)&&pget(s,sc,qx,qy)&&!next[qy][qx]){next[qy][qx]=1;changed=1;}}
                    }
                    memcpy(carry,next,sizeof(carry));if(!changed)break;
                }
                for(int y=0;y<s->h;y++)for(int x=0;x<s->w;x++)if(pget(s,sc,x,y)&&!carry[y][x]){
                    int nx=x+dx,ny=y+dy;if(in(s,nx,ny))score[ny][nx]+=amp;
                }
            }
        }
        /* The only discrete boundary for D2 birth: target voxel measurement. */
        for(int y=0;y<s->h;y++)for(int x=0;x<s->w;x++)if(score[y][x]>=0.5)por(out,tc,x,y,1);
    }
}

static void learn_event_ch(Field*f,const State*s,int a,const State*n,int ch){
    int tx0[128],ty0[128],x0[128],y0[128],ncur=collectp(s,ch,x0,y0,128);
    int nt0=(ncur>0)?learned_targets(f,s,a,ch,tx0,ty0,NORMAL):-1;

    if(ncur>0 && all_mass_waves_closed(s,n)){
        int mx0,my0,mx1,my1;address_anchor(s,ch,&mx0,&my0);address_anchor(n,ch,&mx1,&my1);
        int gx=mx1-mx0,gy=my1-my0;
        uint64_t pi[32],po[32],uni[64];int ni=local_pair_waves(s,ch,pi,32),no=local_pair_waves(n,ch,po,32),nu=0;for(int i=0;i<ni;i++)add_u64(uni,&nu,64,pi[i]);for(int i=0;i<no;i++)add_u64(uni,&nu,64,po[i]);
        double residual=fabs((double)gx)+fabs((double)gy);for(int i=0;i<nu;i++)residual+=fabs((double)has_u64(pi,ni,uni[i])-(double)has_u64(po,no,uni[i]));
        if(residual>0){double wt=residual/(1.0+residual);if(abs(gx)<=7&&abs(gy)<=7)learn_shift_output(f,a,ch,gx,gy,wt);for(int i=0;i<ni;i++)learn_pair_outputs(f,a,pi[i],po,no,wt);}
    }

    if(nt0==ncur && ncur>0){
        for(int i=0;i<ncur;i++){
            int ib=0,mask=rawmask(s,tx0[i],ty0[i],&ib);
            int freeok=ib&&pget(n,ch,tx0[i],ty0[i]);int ov=!freeok;
            uint64_t ctx=CTX_MASK(ch,ib,mask),ovw=OVERFLOW(ch);upd(f,a,ctx,ovw,1,ov);
            if(ov){int nib=0,nmask=rawmask(n,x0[i],y0[i],&nib);(void)nib;uint64_t actual=RAWCTX_MASK(1,nmask);for(int qm=0;qm<(1<<PLANES);qm++){uint64_t cand=RAWCTX_MASK(1,qm);upd(f,a,ovw,cand,1,cand==actual);}}
        }
    }
}
static void learn(Field*f,const State*s,int a,const State*n){for(int ch=0;ch<PLANES;ch++)learn_event_ch(f,s,a,n,ch);}
static int overflow_for(Field*f,int a,int ch,int ib,int mask,Mode m){if(m==OVERFLOW_OFF)return 0;int aa=m==FIELD_SHIFT?(a+1)%ACTS:a;return bit(f,aa,CTX_MASK(ch,ib,mask),OVERFLOW(ch));}
static int overflow_fate(Field*f,int a,int ch,Mode m,int out[PLANES]){
    int aa=m==FIELD_SHIFT?(a+1)%ACTS:a;double z=0,zch[PLANES]={0};
    for(int mask=0;mask<(1<<PLANES);mask++){
        double w=posw(f,aa,OVERFLOW(ch),RAWCTX_MASK(1,mask));z+=w;
        for(int k=0;k<PLANES;k++)zch[k]+=w*((mask>>k)&1);
    }
    if(z<=0)return 0;
    for(int k=0;k<PLANES;k++)out[k]=(zch[k]/z)>=0.5;
    return 1;
}
static void predict_obs(Field*f,const State*s,int a,State*n,Mode m){if(m==FIELD_OFF){*n=*s;return;}zero(n,s->w,s->h);for(int ch=0;ch<PLANES;ch++){int x[128],y[128],tx[128],ty[128],na=collectp(s,ch,x,y,128);if(na<=0)continue;int nt=learned_targets(f,s,a,ch,tx,ty,m);if(nt!=na){for(int i=0;i<na;i++)por(n,ch,x[i],y[i],1);continue;}uint8_t carry[MAXH][MAXW];memset(carry,0,sizeof(carry));for(int i=0;i<na;i++){int ib=0,mask=rawmask(s,tx[i],ty[i],&ib);if(overflow_for(f,a,ch,ib,mask,m))carry[y[i]][x[i]]=1;}
        /* Local binary carry relaxation.  No any-object gate: an unclosed voxel
           can influence only occupied neighbors connected by relation waves
           physically present in the current state. */
        int rdx[8],rdy[8],nd=state_unit_dirs(s,ch,rdx,rdy,8);for(int pass=0;pass<s->w*s->h;pass++){uint8_t next[MAXH][MAXW];memcpy(next,carry,sizeof(next));int changed=0;for(int i=0;i<na;i++)if(carry[y[i]][x[i]])for(int r=0;r<nd;r++){int nx=x[i]+rdx[r],ny=y[i]+rdy[r];if(in(s,nx,ny)&&pget(s,ch,nx,ny)&&!next[ny][nx]){next[ny][nx]=1;changed=1;}}memcpy(carry,next,sizeof(carry));if(!changed)break;}
        int outv[PLANES]={0},have_fate=overflow_fate(f,a,ch,m,outv);for(int i=0;i<na;i++){if(carry[y[i]][x[i]]){if(have_fate){for(int k=0;k<PLANES;k++)por(n,k,x[i],y[i],outv[k]);}else por(n,ch,x[i],y[i],1);}else if(in(s,tx[i],ty[i]))por(n,ch,tx[i],ty[i],1);}}}

/* Closure directions are not pre-enumerated. They are the unit relation waves
   physically present in the current state. */
static int state_unit_dirs(const State*s,int ch,int dx[],int dy[],int cap){
    int n=0;
    for(int y=0;y<s->h;y++)for(int x=0;x<s->w;x++)if(pget(s,ch,x,y)){
        for(int oy=-1;oy<=1;oy++)for(int ox=-1;ox<=1;ox++){
            if(abs(ox)+abs(oy)!=1)continue;
            int nx=x+ox,ny=y+oy;if(!in(s,nx,ny)||!pget(s,ch,nx,ny))continue;
            int seen=0;for(int i=0;i<n;i++)seen|=(dx[i]==ox&&dy[i]==oy);
            if(!seen&&n<cap){dx[n]=ox;dy[n]=oy;n++;}
        }
    }
    return n;
}
/* Direction-symmetric physical boundary carry.  Opposite directions generate
   the same closure-wave identity; readout deduplicates that wave identity. */
static int boundary_carry_seed_dir(const State*s,int ch,int dx,int dy,uint8_t seed[MAXH][MAXW]){
    uint8_t reach[MAXH][MAXW],next[MAXH][MAXW],ack[MAXH][MAXW];
    memset(reach,0,sizeof(reach));memset(seed,0,MAXH*MAXW*sizeof(uint8_t));
    /* All source-boundary occupied voxels emit the same binary carry wave. */
    for(int y=0;y<s->h;y++)for(int x=0;x<s->w;x++)
        if(pget(s,ch,x,y)&&!in(s,x-dx,y-dy))reach[y][x]=1;
    for(int st=0;st<s->w+s->h;st++){
        memcpy(next,reach,sizeof(next));int changed=0;
        for(int y=0;y<s->h;y++)for(int x=0;x<s->w;x++)if(reach[y][x]){
            int nx=x+dx,ny=y+dy;
            if(in(s,nx,ny)&&pget(s,ch,nx,ny)&&!next[ny][nx]){next[ny][nx]=1;changed=1;}
        }
        memcpy(reach,next,sizeof(reach));if(!changed)break;
    }
    memset(ack,0,sizeof(ack));int closed=0;
    for(int y=0;y<s->h;y++)for(int x=0;x<s->w;x++)
        if(reach[y][x]&&!in(s,x+dx,y+dy)){ack[y][x]=1;closed=1;}
    if(!closed)return 0;
    /* Opposite-boundary echo propagates back only through the forward wave.
       No endpoint selection, row index, path list, or visited traversal. */
    for(int st=0;st<s->w+s->h;st++){
        memcpy(next,ack,sizeof(next));int changed=0;
        for(int y=0;y<s->h;y++)for(int x=0;x<s->w;x++)if(ack[y][x]){
            int px=x-dx,py=y-dy;
            if(in(s,px,py)&&reach[py][px]&&pget(s,ch,px,py)&&!next[py][px]){next[py][px]=1;changed=1;}
        }
        memcpy(ack,next,sizeof(ack));if(!changed)break;
    }
    memcpy(seed,ack,MAXH*MAXW*sizeof(uint8_t));return 1;
}
static int boundary_carry_closed_dir(const State*s,int ch,int dx,int dy){uint8_t seed[MAXH][MAXW];return boundary_carry_seed_dir(s,ch,dx,dy,seed);}

typedef struct { uint64_t w; int dx,dy; } LocalRel;
static int add_local_rel(LocalRel r[],int*n,int cap,int ch,int dx,int dy){

}
static int learned_closure_relations(Field*f,int act,int ch,uint64_t iw,LocalRel r[],int cap){

/* Generic residual-to-relation birth.  This learner knows no row/span/closure
   geometry.  Given any already-grounded residual context wave, it asks which
   local address displacement best explains the pixels that actually changed
   between the model's intermediate prediction and reality. */
static int residual_relation_candidates(const State*mid,const State*truth,int ch,LocalRel r[],int cap){
    int n=0;
    for(int y=0;y<mid->h;y++)for(int x=0;x<mid->w;x++){
        int mv=pget(mid,ch,x,y),tv=pget(truth,ch,x,y);if(mv==tv||!tv)continue;
        for(int sy=y-1;sy<=y+1;sy++)for(int sx=x-1;sx<=x+1;sx++){
            if(!in(mid,sx,sy)||!pget(mid,ch,sx,sy))continue;
            add_local_rel(r,&n,cap,ch,x-sx,y-sy);
        }
    }
    return n;
}
static void learn_generic_residual_relations(Field*f,const State*mid,const State*truth,int act,int ch,uint64_t iw){
    LocalRel cand[16];int nc=residual_relation_candidates(mid,truth,ch,cand,16);if(nc<=0)return;
    int nr=0;for(int y=0;y<mid->h;y++)for(int x=0;x<mid->w;x++)nr+=pget(mid,ch,x,y)!=pget(truth,ch,x,y);if(nr<=0)return;
    for(int j=0;j<nc;j++){
        double sc=0;
        for(int y=0;y<mid->h;y++)for(int x=0;x<mid->w;x++)if(pget(mid,ch,x,y)!=pget(truth,ch,x,y)){

        }

}

static int closure_relax_once(Field*f,const State*s,int act,State*out){
    *out=*s;int changed=0;
    for(int ch=0;ch<PLANES;ch++){
        int cdxv[8],cdyv[8],nd=state_unit_dirs(s,ch,cdxv,cdyv,8);uint64_t seen[8]={0};int nseen=0;
        for(int d=0;d<nd;d++){
            int cdx=cdxv[d],cdy=cdyv[d];uint8_t seed[MAXH][MAXW];if(!boundary_carry_seed_dir(s,ch,cdx,cdy,seed))continue;
            uint64_t iw=CLOSURE_WAVE_DIR(ch,cdx,cdy);int dup=0;for(int j=0;j<nseen;j++)dup|=(seen[j]==iw);if(dup)continue;if(nseen<8)seen[nseen++]=iw;
            LocalRel cand[16];int nc=learned_closure_relations(f,act,ch,iw,cand,16);if(nc<=0)continue;
            double sx=0,sy=0,z=0;for(int j=0;j<nc;j++){double w=posw(f,act,iw,cand[j].w);sx+=w*cand[j].dx;sy+=w*cand[j].dy;z+=w;}
            if(z<=0)continue;
            double mag=hypot(sx,sy);
            if(mag<=1e-12)continue;
            int dx=(int)lround(sx/mag),dy=(int)lround(sy/mag);
            /* No Manhattan-only child gate. Any reality-born local relation in
               the radius-1 physical neighborhood may act. (0,0) is the natural
               identity and therefore has no state-changing relaxation.) */
            if(dx==0&&dy==0)continue;
            if(abs(dx)>1||abs(dy)>1)continue;
            uint8_t tmp[MAXH][MAXW],holes[MAXH][MAXW],next[MAXH][MAXW];memset(tmp,0,sizeof(tmp));memcpy(holes,seed,sizeof(holes));
            for(int y=0;y<s->h;y++)for(int x=0;x<s->w;x++)tmp[y][x]=pget(s,ch,x,y);
            while(mask_any(s,holes)){
                uint8_t nh[MAXH][MAXW],nb[MAXH][MAXW];memcpy(nb,tmp,sizeof(nb));memset(nh,0,sizeof(nh));int any=0;
                for(int y=0;y<s->h;y++)for(int x=0;x<s->w;x++)if(holes[y][x]){
                    int px=x-dx,py=y-dy;int v=0;
                    if(in(s,px,py)){
                        if(holes[py][px]){v=0;nh[y][x]=1;any=1;}
                        else{v=tmp[py][px];nh[py][px]=1;any=1;}
                    }
                    nb[y][x]=(uint8_t)v;
                }
                memcpy(tmp,nb,sizeof(tmp));memcpy(next,nh,sizeof(next));memcpy(holes,next,sizeof(holes));if(!any)break;
            }
            for(int y=0;y<s->h;y++)for(int x=0;x<s->w;x++){uint8_t old=pget(out,ch,x,y);pset(out,ch,x,y,tmp[y][x]);changed|=(old!=tmp[y][x]);}
        }
    }
    return changed;
}

/* Relation-pair waves are reversible physical identities, not enumerated roles.
   Only one-hop relations that actually coexist in a current local voxel state
   can create a pair wave. */
static int local_relation_ok(int dx,int dy){return abs(dx)+abs(dy)==1;}
static unsigned relation_key(int dx,int dy){return (unsigned)(((dx+8)&15)<<4)|((unsigned)((dy+8)&15));}
static uint64_t pair_wave(int dx1,int dy1,int dx2,int dy2){
    if(!local_relation_ok(dx1,dy1)||!local_relation_ok(dx2,dy2)||(dx1==dx2&&dy1==dy2))return 0;
    unsigned a=relation_key(dx1,dy1),b=relation_key(dx2,dy2);
    if(a>b){int tx=dx1,ty=dy1;dx1=dx2;dy1=dy2;dx2=tx;dy2=ty;}
    return (3ULL<<62)|((uint64_t)((dx1+8)&15)<<12)|((uint64_t)((dy1+8)&15)<<8)|((uint64_t)((dx2+8)&15)<<4)|(uint64_t)((dy2+8)&15);
}
static int decode_pair_wave(uint64_t w,int*dx1,int*dy1,int*dx2,int*dy2){
    if((w>>62)!=3)return 0;
    *dx1=(int)((w>>12)&15)-8;*dy1=(int)((w>>8)&15)-8;
    *dx2=(int)((w>>4)&15)-8;*dy2=(int)(w&15)-8;
    return local_relation_ok(*dx1,*dy1)&&local_relation_ok(*dx2,*dy2)&&!(*dx1==*dx2&&*dy1==*dy2);
}

        int x=99,y=99;if(ax==rdx&&ay==rdy){x=bx;y=by;}else if(bx==rdx&&by==rdy){x=ax;y=ay;}else continue;
        int seen=0;for(int j=0;j<nq;j++)seen|=(qdx[j]==x&&qdy[j]==y);if(!seen&&nq<16){qdx[nq]=x;qdy[nq]=y;nq++;}
    }
    for(int q=0;q<nq;q++){
        uint64_t iw=pair_wave(rdx,rdy,qdx[q],qdy[q]);if(!iw)continue;uint64_t aiw=collide(ACTION_WAVE(aa),iw),outs[64];int no=0;
        for(int i=0;i<f->n&&no<64;i++)if(f->c[i].inw==iw||f->c[i].inw==aiw){int ax,ay,bx,by;if(!decode_pair_wave(f->c[i].outw,&ax,&ay,&bx,&by))continue;add_u64(outs,&no,64,f->c[i].outw);}
        double localz=0;int ldx[16],ldy[16],nl=0;double ls[16]={0};
        for(int i=0;i<no;i++){int ax=0,ay=0,bx=0,by=0;if(!decode_pair_wave(outs[i],&ax,&ay,&bx,&by))continue;double w=posw(f,aa,iw,outs[i]);if(w<=0)continue;add_dir_score(ldx,ldy,ls,&nl,16,ax,ay,w);add_dir_score(ldx,ldy,ls,&nl,16,bx,by,w);localz+=w;}
        if(localz>0){for(int i=0;i<nl;i++)add_dir_score(sdx,sdy,score,&ns,16,ldx[i],ldy[i],ls[i]/localz);evidence++;}
    }
    if(!evidence||ns<=0)return 0;
    double sx=0,sy=0;for(int i=0;i<ns;i++){sx+=score[i]*sdx[i];sy+=score[i]*sdy[i];}double mag=hypot(sx,sy);if(mag<=1e-9)return 0;*outx=sx/mag;*outy=sy/mag;return 1;
}
static void address_anchor(const State*s,int ch,int*ax,int*ay){uint8_t v[MAXH][MAXW],h[MAXH][MAXW];memset(v,0,sizeof(v));memset(h,0,sizeof(h));for(int y=0;y<s->h;y++)for(int x=0;x<s->w;x++)v[y][x]=h[y][x]=!!pget(s,ch,x,y);for(int st=0;st<s->h;st++)for(int y=1;y<s->h;y++)for(int x=0;x<s->w;x++)v[y-1][x]|=v[y][x];for(int st=0;st<s->w;st++)for(int x=1;x<s->w;x++)for(int y=0;y<s->h;y++)h[y][x-1]|=h[y][x];int x=0;while(x<s->w&&!v[0][x])x++;int y=0;while(y<s->h&&!h[y][0])y++;*ax=x;*ay=y;}
static int learned_targets(Field*f,const State*s,int a,int ch,int tx[],int ty[],Mode m){
    int x[128],y[128],n=collectp(s,ch,x,y,128);if(n<=0||n>128)return -1;
    int gx,gy,anchorx,anchory;address_anchor(s,ch,&anchorx,&anchory);if(!learned_shift(f,a,ch,&gx,&gy,m))return -1;
    int rdx[8],rdy[8],nd=current_unit_dirs(s,ch,rdx,rdy,8);double mdx[8],mdy[8];
    for(int r=0;r<nd;r++)if(!mapped_vec_local(f,a,rdx[r],rdy[r],&mdx[r],&mdy[r],m))return -1;
    int idx[MAXH][MAXW];for(int yy=0;yy<MAXH;yy++)for(int xx=0;xx<MAXW;xx++)idx[yy][xx]=-1;for(int i=0;i<n;i++)idx[y[i]][x[i]]=i;
    double ox[128]={0},oy[128]={0},nxv[128],nyv[128];
    int relax_safety=s->w*s->h*8;for(int pass=0;pass<relax_safety;pass++){
        double maxd=0;
        for(int i=0;i<n;i++){
            double sx=0,sy=0,z=0;
            for(int r=0;r<nd;r++){
                int xx=x[i]+rdx[r],yy=y[i]+rdy[r];if(!in(s,xx,yy))continue;int j=idx[yy][xx];if(j<0)continue;
                sx+=ox[j]-mdx[r];sy+=oy[j]-mdy[r];z+=1.0;
            }
            if(z>0){double tx0=sx/z,ty0=sy/z;nxv[i]=0.5*ox[i]+0.5*tx0;nyv[i]=0.5*oy[i]+0.5*ty0;}else{nxv[i]=ox[i];nyv[i]=oy[i];}
        }
        double mx=0,my=0;for(int i=0;i<n;i++){mx+=nxv[i];my+=nyv[i];}mx/=n;my/=n;
        for(int i=0;i<n;i++){nxv[i]-=mx;nyv[i]-=my;double d=fabs(nxv[i]-ox[i])+fabs(nyv[i]-oy[i]);if(d>maxd)maxd=d;ox[i]=nxv[i];oy[i]=nyv[i];}
        if(maxd<1e-8)break;
    }
    double mnx=ox[0],mny=oy[0];for(int i=1;i<n;i++){if(ox[i]<mnx)mnx=ox[i];if(oy[i]<mny)mny=oy[i];}
    for(int i=0;i<n;i++){tx[i]=anchorx+gx+(int)lround(ox[i]-mnx);ty[i]=anchory+gy+(int)lround(oy[i]-mny);}return n;
}
typedef struct { uint8_t b[16]; } MassWave;
static void mass_inject_one(MassWave*m){int carry=1;for(int i=0;i<16&&carry;i++){int old=m->b[i];m->b[i]=(uint8_t)(old^carry);carry=old&carry;}}
static MassWave information_mass_wave(const State*s,int ch){MassWave m;memset(&m,0,sizeof(m));for(int y=0;y<s->h;y++)for(int x=0;x<s->w;x++)if(pget(s,ch,x,y))mass_inject_one(&m);return m;}
static uint64_t mass_state_wave(const State*s,int ch){
    MassWave m=information_mass_wave(s,ch);uint64_t w=PLANE(ch);
    for(int i=0;i<16;i++)w=collide(w,prim(5,5,i,m.b[i]?1:0,3));
    return w;
}
static uint64_t cross_mass_context(const State*s,int tc,int sc){
    uint64_t a=mass_state_wave(s,tc),b=PLANE(sc);if(a>b){uint64_t t=a;a=b;b=t;}return collide(a,b);
}
static int mass_wave_closed(const State*a,const State*b,int ch){MassWave x=information_mass_wave(a,ch),y=information_mass_wave(b,ch);for(int i=0;i<16;i++)if(x.b[i]!=y.b[i])return 0;return 1;}
static int all_mass_waves_closed(const State*a,const State*b){
    for(int ch=0;ch<PLANES;ch++)if(!mass_wave_closed(a,b,ch))return 0;
    return 1;
}
/* v0.181: generic consequence credit for cross-channel function birth.
   No equal-mass gate and no shape-coherence classifier.  A translated source
   wave receives credit only to the extent that adding its predicted pixels
   reduces the actual target-channel residual. */
static double cross_residual_credit(const State*mid,const State*truth,int sc,int tc,int dx,int dy){
    int before=0,after=0;
    for(int y=0;y<mid->h;y++)for(int x=0;x<mid->w;x++){
        int real=!!pget(truth,tc,x,y),base=!!pget(mid,tc,x,y);
        before+=(base!=real);
        int sx=x-dx,sy=y-dy;
        int add=(in(mid,sx,sy)&&pget(mid,sc,sx,sy));
        int pred=base||add;
        after+=(pred!=real);
    }
    if(before<=0||after>=before)return 0.0;
    return (double)(before-after)/(double)before;
}
static void learn_mass_context_birth(Field*f,const State*mid,const State*truth,int act){
    for(int tc=0;tc<PLANES;tc++){
        if(mass_wave_closed(mid,truth,tc))continue;
        for(int sc=0;sc<PLANES;sc++)if(sc!=tc){
            int sx,sy,tx,ty;address_anchor(mid,sc,&sx,&sy);address_anchor(truth,tc,&tx,&ty);
            int dx=tx-sx,dy=ty-sy;if(abs(dx)>7||abs(dy)>7)continue;
            double credit=cross_residual_credit(mid,truth,sc,tc,dx,dy);if(credit<=0.0)continue;
            uint64_t iw=cross_mass_context(mid,tc,sc),actual=XREL(sc,tc,dx,dy);
            upd(f,act,iw,actual,credit,1);
        }
    }
}
static void mass_context_birth_relax(Field*f,const State*s,int act,State*out){
    *out=*s;
    for(int tc=0;tc<PLANES;tc++){
        double score[MAXH][MAXW];memset(score,0,sizeof(score));
        for(int sc=0;sc<PLANES;sc++)if(sc!=tc){
            uint64_t iw=cross_mass_context(s,tc,sc),aiw=collide(ACTION_WAVE(act),iw),outs[64];int no=0;
            for(int i=0;i<f->n&&no<64;i++)if(f->c[i].inw==iw||f->c[i].inw==aiw){int dx,dy;if(!is_xrel(f->c[i].outw,sc,tc,&dx,&dy))continue;int seen=0;for(int j=0;j<no;j++)seen|=outs[j]==f->c[i].outw;if(!seen)outs[no++]=f->c[i].outw;}
            for(int oi=0;oi<no;oi++){
                int dx=0,dy=0;if(!is_xrel(outs[oi],sc,tc,&dx,&dy))continue;
                /* Every D2 relation wave participates continuously.  No
                   internal birth threshold and no winner direction.  Its
                   positive amplitude is converted to physical contribution;
                   discreteness happens only when target voxels are measured. */
                double amp=2.0*posw(f,act,iw,outs[oi]);if(amp<=0)continue;
                uint8_t carry[MAXH][MAXW];memset(carry,0,sizeof(carry));
                for(int y=0;y<s->h;y++)for(int x=0;x<s->w;x++)if(pget(s,sc,x,y)){
                    int nx=x+dx,ny=y+dy,ib=0;int mask=rawmask(s,nx,ny,&ib);
                    if(bit(f,act,CTX_MASK(tc,ib,mask),OVERFLOW(tc)))carry[y][x]=1;
                }
                int rdx[8],rdy[8],nd=state_unit_dirs(s,sc,rdx,rdy,8);
                for(int pass=0;pass<s->w*s->h;pass++){
                    uint8_t next[MAXH][MAXW];memcpy(next,carry,sizeof(next));int changed=0;
                    for(int y=0;y<s->h;y++)for(int x=0;x<s->w;x++)if(pget(s,sc,x,y)&&carry[y][x]){
                        for(int r=0;r<nd;r++){int qx=x+rdx[r],qy=y+rdy[r];if(in(s,qx,qy)&&pget(s,sc,qx,qy)&&!next[qy][qx]){next[qy][qx]=1;changed=1;}}
                    }
                    memcpy(carry,next,sizeof(carry));if(!changed)break;
                }
                for(int y=0;y<s->h;y++)for(int x=0;x<s->w;x++)if(pget(s,sc,x,y)&&!carry[y][x]){
                    int nx=x+dx,ny=y+dy;if(in(s,nx,ny))score[ny][nx]+=amp;
                }
            }
        }
        /* The only discrete boundary for D2 birth: target voxel measurement. */
        for(int y=0;y<s->h;y++)for(int x=0;x<s->w;x++)if(score[y][x]>=0.5)por(out,tc,x,y,1);
    }
}
