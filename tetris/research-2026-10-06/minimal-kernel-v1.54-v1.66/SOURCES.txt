===== bpc_minimal_v154_reality_tick.c =====
#include <stdio.h>
#include <stdint.h>
#include <string.h>

#define PORTS 24
#define CAP 1024
#define MAXDELTA 2048

typedef struct{uint64_t key;double off,on;uint8_t used;}Cell;
typedef struct{Cell c[CAP];int n;}Field;
typedef struct{uint8_t bit[PORTS];}Medium;
typedef struct{int s,d;double dm;}Delta;
static uint64_t rs=0x154082efa98ec4e6ULL;
static uint64_t ru(void){rs^=rs>>12;rs^=rs<<25;rs^=rs>>27;return rs*2685821657736338717ULL;}
static uint64_t rkey(int s,int d){return((uint64_t)(uint32_t)s<<32)|(uint32_t)d;}

/* The only state-change primitive. */
static void move_mass(double *off,double *on,double dm){
    if(dm>0){if(dm>*off)dm=*off;*off-=dm;*on+=dm;}
    else if(dm<0){double q=-dm;if(q>*on)q=*on;*on-=q;*off+=q;}
}
static int idx(const Field*f,int s,int d){uint64_t k=rkey(s,d);for(int i=0;i<f->n;i++)if(f->c[i].used&&f->c[i].key==k)return i;return-1;}
static double amp(const Field*f,int s,int d){int i=idx(f,s,d);return i>=0?f->c[i].on:0.0;}
static Cell* ensure(Field*f,int s,int d){
    int i=idx(f,s,d);if(i>=0)return &f->c[i];
    for(int j=0;j<f->n;j++)if(!f->c[j].used){f->c[j]=(Cell){rkey(s,d),1.0,0.0,1};return &f->c[j];}
    if(f->n>=CAP)return NULL;
    f->c[f->n]=(Cell){rkey(s,d),1.0,0.0,1};return &f->c[f->n++];
}
static void cleanup(Field*f){for(int i=0;i<f->n;i++)if(f->c[i].used&&f->c[i].on<=0.0)f->c[i].used=0;}
static double relation_output(const Field*f,const Medium*m,int d){
    double off=1.0,on=0.0;
    for(int s=0;s<PORTS;s++)if(m->bit[s]){double a=amp(f,s,d);move_mass(&off,&on,off*a);}
    return on;
}

/* v1.54: one reality tick. There is no observe() API and no predict() API.
   The current medium propagates once, the physical reality arrives, one global
   residual writes back along all participating ordered relations, and reality
   then becomes the medium for the next tick. Time exists only through re-entry. */
static int reality_tick(Field*f,Medium*m,const Medium*reality){
    double pred[PORTS];Delta ds[MAXDELTA];int nd=0;int exact=1;
    for(int d=0;d<PORTS;d++){
        pred[d]=relation_output(f,m,d);
        int measured=(pred[d]>.5);
        if(measured!=reality->bit[d])exact=0;
        double e=(double)reality->bit[d]-pred[d];
        if(e==0.0)continue;
        for(int s=0;s<PORTS;s++)if(m->bit[s]&&nd<MAXDELTA)ds[nd++]=(Delta){s,d,e};
    }
    for(int i=0;i<nd;i++){
        int j=idx(f,ds[i].s,ds[i].d);Cell*q=j>=0?&f->c[j]:NULL;
        if(!q&&ds[i].dm>0)q=ensure(f,ds[i].s,ds[i].d);
        if(q)move_mass(&q->off,&q->on,ds[i].dm);
    }
    cleanup(f);*m=*reality;return exact;
}
static int resident(const Field*f){int n=0;for(int i=0;i<f->n;i++)n+=f->c[i].used;return n;}
static uint64_t field_hash(const Field*f){uint64_t h=1469598103934665603ULL;for(int i=0;i<f->n;i++)if(f->c[i].used){uint64_t q=f->c[i].key;memcpy(&q,&f->c[i].on,sizeof(double)<sizeof(q)?sizeof(double):sizeof(q));h^=f->c[i].key;h*=1099511628211ULL;uint64_t z;memcpy(&z,&f->c[i].on,sizeof(z));h^=z;h*=1099511628211ULL;}return h;}
static int m1[PORTS],m2[PORTS];
static void init(void){for(int i=0;i<PORTS;i++){m1[i]=(7*i+5)%PORTS;m2[i]=(11*i+3)%PORTS;}}
static void physical_reality(const Medium*m,Medium*r){memset(r,0,sizeof(*r));for(int i=0;i<PORTS;i++)if(m->bit[i]){r->bit[m1[i]]=1;r->bit[m2[i]]=1;}}
static void random_medium(Medium*m,int minbits,int span){memset(m,0,sizeof(*m));int bits=minbits+(int)(ru()%(uint64_t)span),n=0;while(n<bits){int s=(int)(ru()%PORTS);if(!m->bit[s]){m->bit[s]=1;n++;}}}
int main(void){
    init();Field f={0};Medium m={0},r;
    random_medium(&m,2,7);
    /* one continuous stream; no epochs and no separate learner call */
    for(int t=0;t<120000;t++){
        physical_reality(&m,&r);(void)reality_tick(&f,&m,&r);
        /* external perturbation is ordinary reality, not a training mode */
        if((t%3)==2)random_medium(&m,2,7);
    }
    uint64_t h0=field_hash(&f);unsigned long long n=300000,ok=0;
    for(unsigned long long q=0;q<n;q++){
        random_medium(&m,2,7);physical_reality(&m,&r);
        ok+=(unsigned long long)reality_tick(&f,&m,&r);
    }
    uint64_t h1=field_hash(&f);
    double true_min=1.0,false_max=0.0;
    for(int s=0;s<PORTS;s++)for(int d=0;d<PORTS;d++){
        double a=amp(&f,s,d);int truth=(d==m1[s]||d==m2[s]);
        if(truth&&a<true_min)true_min=a;
        if(!truth&&a>false_max)false_max=a;
    }
    printf("V154_REALITY_TICK exact=%llu/%llu=%.6f%% resident=%d true_min=%.6f false_max=%.6f field_unchanged=%d\n",ok,n,100.0*(double)ok/n,resident(&f),true_min,false_max,h0==h1);
    return ok==n&&h0==h1?0:1;
}
