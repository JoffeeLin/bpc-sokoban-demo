#include <stdio.h>
#include <stdint.h>
#include <string.h>
#include <math.h>
#define SCREEN 300
#define ACTS 5
#define N (SCREEN+ACTS)
#define MAXE 96
static uint64_t rs=0x9216d5d98979fb1bULL;
static uint64_t ru(void){rs^=rs>>12;rs^=rs<<25;rs^=rs>>27;return rs*2685821657736338717ULL;}
static double rd(void){return (double)(ru()>>11)*(1.0/9007199254740992.0);}
typedef struct{int src,dst;double w;} Edge;
/* old: two output heads, screen and action ports */
static void old_apply(const double src[N],const Edge*e,int ne,double screen[SCREEN],double action[ACTS]){double ms[SCREEN],ma[ACTS];for(int i=0;i<SCREEN;i++)ms[i]=1;for(int i=0;i<ACTS;i++)ma[i]=1;for(int r=0;r<ne;r++){double a=src[e[r].src]*e[r].w;if(a<=0)continue;if(e[r].dst<SCREEN)ms[e[r].dst]*=1-a;else ma[e[r].dst-SCREEN]*=1-a;}for(int i=0;i<SCREEN;i++)screen[i]=1-ms[i];for(int i=0;i<ACTS;i++)action[i]=1-ma[i];}
/* new: one medium, destinations are just relation addresses */
static void medium_apply(const double src[N],const Edge*e,int ne,double out[N]){double miss[N];for(int i=0;i<N;i++)miss[i]=1;for(int r=0;r<ne;r++){double a=src[e[r].src]*e[r].w;if(a>0)miss[e[r].dst]*=1-a;}for(int i=0;i<N;i++)out[i]=1-miss[i];}
int main(void){unsigned long long n=500000,ok=0;for(unsigned long long t=0;t<n;t++){double src[N];for(int i=0;i<N;i++)src[i]=((ru()%100)<10)?rd():0;Edge e[MAXE];int ne=1+(int)(ru()%MAXE);for(int r=0;r<ne;r++){e[r].src=(int)(ru()%N);e[r].dst=(int)(ru()%N);e[r].w=rd();}double s[SCREEN],a[ACTS],u[N];old_apply(src,e,ne,s,a);medium_apply(src,e,ne,u);int same=1;for(int i=0;i<SCREEN;i++)if(s[i]!=u[i])same=0;for(int i=0;i<ACTS;i++)if(a[i]!=u[SCREEN+i])same=0;if(same)ok++;else{printf("FAIL %llu\n",t);return 1;}}printf("WORLD_ACTION_OUTPUTS_TO_ONE_RELATION_MEDIUM %llu/%llu = %.6f%%\n",ok,n,100.0*(double)ok/n);return 0;}
#include <stdio.h>
#include <stdint.h>
#include <string.h>
#define SRC 8
#define WORLD_OUT 8
#define ACT_OUT 4
#define DST (WORLD_OUT+ACT_OUT)
typedef struct{double yes,total;} Cell;
typedef struct{Cell q[SRC][DST];} Field;
static uint64_t rs=0xd1310ba698dfb5acULL;
static uint64_t ru(void){rs^=rs>>12;rs^=rs<<25;rs^=rs>>27;return rs*2685821657736338717ULL;}
static double prob(const Cell*c){return(c->yes+1.0)/(c->total+2.0);}
static void credit(Cell*c,int target){c->total+=1.0;c->yes+=(target!=0);}
/* one generic residual writer: destination is just an address; no world/action class */
static void observe(Field*f,const uint8_t src[SRC],const uint8_t target[DST]){for(int i=0;i<SRC;i++)if(src[i])for(int d=0;d<DST;d++)credit(&f->q[i][d],target[d]);}
static void predict(const Field*f,const uint8_t src[SRC],uint8_t out[DST]){for(int d=0;d<DST;d++){double miss=1;int any=0;for(int i=0;i<SRC;i++)if(src[i]){any=1;double p=prob(&f->q[i][d]);double a=2*p-1;if(a<0)a=0;if(a>1)a=1;miss*=1-a;}out[d]=(uint8_t)(any&&(1-miss>.5));}}
static int wmap[SRC]={1,3,5,7,0,2,4,6};
static int amap[SRC]={0,1,2,3,1,2,3,0};
static void teacher(const uint8_t src[SRC],uint8_t out[DST]){memset(out,0,DST);for(int i=0;i<SRC;i++)if(src[i]){out[wmap[i]]=1;out[WORLD_OUT+amap[i]]=1;}}
int main(void){Field f={0};/* training: exactly one active source per experience; no multi-source combinations */for(int ep=0;ep<5000;ep++)for(int i=0;i<SRC;i++){uint8_t s[SRC]={0},t[DST];s[i]=1;teacher(s,t);observe(&f,s,t);}unsigned long long n=100000,ok=0;for(unsigned long long q=0;q<n;q++){uint8_t s[SRC]={0},t[DST],p[DST];int bits=2+(int)(ru()%5);for(int k=0;k<bits;k++)s[ru()%SRC]=1;teacher(s,t);predict(&f,s,p);if(!memcmp(t,p,DST))ok++;else{printf("FAIL %llu\n",q);return 1;}}printf("ONE_FIELD_WORLD_AND_ACTION unseen_multi_source=%llu/%llu = %.6f%%\n",ok,n,100.0*(double)ok/n);puts("Training contained only singleton demonstrations; world bits and action ports used the same Cell, credit(), and predict() equations.");return 0;}
