#include <stdio.h>
#include <stdint.h>
#include <string.h>
#include <math.h>
#define W 10
#define H 20
#define N (W*H)
#define MAXN 64
#define MAXE 512

typedef struct{int i,j;double dx,dy;} Edge;
static uint64_t rs=0x452821e638d01377ULL;
static uint64_t ru(void){rs^=rs>>12;rs^=rs<<25;rs^=rs>>27;return rs*2685821657736338717ULL;}

static void old_relax(int n,const Edge*e,int ne,double ox[MAXN],double oy[MAXN]){
  double nx[MAXN],ny[MAXN];
  for(int pass=0;pass<256;pass++){
    double maxd=0;
    for(int i=0;i<n;i++){
      double sx=0,sy=0,z=0;
      for(int k=0;k<ne;k++)if(e[k].i==i){int j=e[k].j;sx+=ox[j]-e[k].dx;sy+=oy[j]-e[k].dy;z+=1;}
      if(z>0){double qx=sx/z,qy=sy/z;nx[i]=0.5*ox[i]+0.5*qx;ny[i]=0.5*oy[i]+0.5*qy;}else{nx[i]=ox[i];ny[i]=oy[i];}
    }
    double mx=0,my=0;for(int i=0;i<n;i++){mx+=nx[i];my+=ny[i];}mx/=n;my/=n;
    for(int i=0;i<n;i++){nx[i]-=mx;ny[i]-=my;double d=fabs(nx[i]-ox[i])+fabs(ny[i]-oy[i]);if(d>maxd)maxd=d;ox[i]=nx[i];oy[i]=ny[i];}
    if(maxd<1e-8)break;
  }
}

/* Generic relation-constraint relaxation: arbitrary payload dimension, directed
   edge transform, synchronous averaging, damping, and zero-mean gauge. */
static void relation_constraint_relax(int n,int dim,const Edge*e,int ne,double state[MAXN][2]){
  double next[MAXN][2];
  for(int pass=0;pass<256;pass++){
    double maxd=0;
    for(int i=0;i<n;i++){
      double sum[2]={0,0},z=0;
      for(int k=0;k<ne;k++)if(e[k].i==i){int j=e[k].j;double tr[2]={e[k].dx,e[k].dy};for(int d=0;d<dim;d++)sum[d]+=state[j][d]-tr[d];z+=1;}
      for(int d=0;d<dim;d++)next[i][d]=(z>0)?0.5*state[i][d]+0.5*(sum[d]/z):state[i][d];
    }
    double mean[2]={0,0};for(int i=0;i<n;i++)for(int d=0;d<dim;d++)mean[d]+=next[i][d];for(int d=0;d<dim;d++)mean[d]/=n;
    for(int i=0;i<n;i++)for(int d=0;d<dim;d++){next[i][d]-=mean[d];double dd=fabs(next[i][d]-state[i][d]);/* old max is L1 per node, handled below */(void)dd;}
    for(int i=0;i<n;i++){double dd=0;for(int d=0;d<dim;d++)dd+=fabs(next[i][d]-state[i][d]);if(dd>maxd)maxd=dd;for(int d=0;d<dim;d++)state[i][d]=next[i][d];}
    if(maxd<1e-8)break;
  }
}

int main(void){unsigned long long trials=100000,ok=0;for(unsigned long long t=0;t<trials;t++){
  int n=1+(int)(ru()%40);Edge e[MAXE];int ne=0;
  /* random directed constraint graph with transforms chosen from local relation space */
  for(int i=0;i<n;i++)for(int j=0;j<n;j++)if(i!=j&&(ru()%100)<10&&ne<MAXE){e[ne].i=i;e[ne].j=j;e[ne].dx=(int)(ru()%3)-1;e[ne].dy=(int)(ru()%3)-1;ne++;}

  int same=1;for(int i=0;i<n;i++)if(ax[i]!=s[i][0]||ay[i]!=s[i][1]){same=0;break;}
  if(same)ok++;else{double md=0;for(int i=0;i<n;i++){double d=fabs(ax[i]-s[i][0])+fabs(ay[i]-s[i][1]);if(d>md)md=d;}printf("FAIL t=%llu n=%d ne=%d max=%.17g\n",t,n,ne,md);return 1;}
 }
