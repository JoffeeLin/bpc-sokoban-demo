static int old_read(const OldGauge*g,int a,int*dx,int*dy){

  }

    sx+=w*x; sy+=w*y; z+=w;

        /* Generic execution conceptually scans every output relation. In the

        vx+=0.0*bb;vy+=1.0*bb;
      }
    }
    double gate=1.0;
    for(int i=0;i<nr;i++)for(int j=i+1;j<nr;j++){

      double ps=ra[i]*ra[j],g=p->pair[x][y];gate*=1.0-ps*(1.0-g);
    }

  }
}

static void new_reach(const uint8_t carrier[N],const uint8_t domain[N],int dx,int dy,uint8_t wave[N]){Edge e={dx,dy,1.0};for(int step=0;step<N;step++){double src[N]={0},arr[N];for(int i=0;i<N;i++)src[i]=wave[i]?1.0:0.0;relation_apply(src,&e,1,arr);int changed=0;for(int i=0;i<N;i++)if(arr[i]>.5&&domain[i]&&carrier[i]&&!wave[i]){wave[i]=1;changed=1;}if(!changed)break;}}
